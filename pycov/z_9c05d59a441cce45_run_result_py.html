25""" 

26The result file of a kpex run, for the scripts that drive kpex, so that they neither hard-code the paths 

27of its outputs nor parse its log (#427): kpex_result.json in the run directory, also written to --out-result. 

28Its keys, and when it is written, are described in 'Result file' of DEVELOPMENT.md. 

29 

30Each write replaces it as a whole (a temporary file, renamed), so that it is never read half-written. 

31FORMAT_VERSION goes up when a key is removed or changes its meaning, not for new keys. 

32"""

90@dataclass(frozen=True) 

91class WrittenFile: 

92 kind: FileKind 

93 path: str # absolute 

94 engine: Optional[ExtractionEngine] = None # None for the files of the run, e.g. its log 

95 netlist_format: Optional[NetlistFormat] = None # of a netlist 

96 reduced: bool = False # of a netlist 

97 ports: Optional[List[str]] = None # of a netlist, as its .SUBCKT line of the cell has them, None without it

99 def to_json(self) -> Dict[str, Any]: 

100 written = {'kind': self.kind.value, 'path': self.path, 

101 'engine': self.engine.value if self.engine is not None else None} 

102 if self.kind in NETLIST_KINDS: 

103 written.update({'format': self.netlist_format.value, 'reduced': self.reduced, 'ports': self.ports}) 

104 return written

115class _LoggedProblems(logging.Handler): 

116 """The warnings and errors logged during the run""" 

117 

118 def __init__(self, result: RunResult): 

119 super().__init__(level=logging.WARNING) 

120 self.result = result 

121 

122 def emit(self, record: logging.LogRecord): 

123 try: 

124 message = record.getMessage() 

125 except Exception: 

126 self.handleError(record) 

127 return 

128 (self.result.errors if record.levelno >= logging.ERROR else self.result.warnings).append(message)

131@dataclass 

132class RunResult: 

133 """ 

134 What the result file has of the run: its callers note what they know, when they know it 

135 """ 

136 command: Optional[str] = None # e.g. 'extract' 

137 arguments: List[str] = field(default_factory=list) # of the command line, after the program name 

138 pdk: Optional[str] = None 

139 cell: Optional[str] = None 

140 engines: List[ExtractionEngine] = field(default_factory=list) 

141 mode: Optional[str] = None # --mode of the 2.5D engine 

142 inputs: Dict[str, Optional[str]] = field(default_factory=dict) # e.g. {'gds': …}, absolute paths 

143 run_dir: Optional[str] = None # absolute 

144 out_result_path: Optional[str] = None # --out-result 

145 stage: Stage = Stage.ARGUMENTS 

146 exit_code: Optional[int] = None # None until the run ends 

147 halo_um: Optional[float] = None 

148 devices: Optional[DeviceCounts] = None 

149 check_problems: Optional[List[str]] = None 

150 device_capacitance_checks: Optional[List[Dict[str, Any]]] = None 

151 files: List[WrittenFile] = field(default_factory=list) 

152 warnings: List[str] = field(default_factory=list) 

153 errors: List[str] = field(default_factory=list) 

154 _announced_paths: Set[str] = field(default_factory=set, repr=False) 

155 _failed_paths: Set[str] = field(default_factory=set, repr=False)

157 def add_file(self, 

158 kind: FileKind, 

159 path: str, 

160 engine: Optional[ExtractionEngine] = None, 

161 netlist_format: Optional[NetlistFormat] = None, 

162 reduced: bool = False, 

163 ports: Optional[List[str]] = None): 

164 """ 

165 :param path: of a file the run wrote 

166 """ 

167 self.files.append(WrittenFile(kind=kind, path=os.path.abspath(path), engine=engine, 

168 netlist_format=netlist_format, reduced=reduced, ports=ports))

170 def enter(self, stage: Stage): 

171 self.stage = stage 

172 self.write()

174 def paths(self) -> List[str]: 

175 """ 

176 :return: where the result file goes: into the run directory once it is known, and to --out-result 

177 """ 

178 paths = [os.path.join(self.run_dir, RESULT_FILE_NAME)] if self.run_dir is not None else [] 

179 if self.out_result_path is not None and os.path.abspath(self.out_result_path) not in paths: 

180 paths.append(os.path.abspath(self.out_result_path)) 

181 return paths

184 memory = peak_memory() 

185 return { 

186 'format_version': FORMAT_VERSION, 

187 'kpex_version': __version__, 

188 'command': self.command, 

189 'arguments': self.arguments, 

190 'pdk': self.pdk, 

191 'cell': self.cell, 

192 'engines': [engine.value for engine in self.engines], 

193 'mode': self.mode, 

194 'inputs': self.inputs, 

195 'run_dir': self.run_dir, 

196 'stage': self.stage.value, 

197 'exit_code': self.exit_code, 

198 'halo_um': self.halo_um, 

199 'devices': dataclasses.asdict(self.devices) if self.devices is not None else None, 

200 'check_problems': self.check_problems, 

201 'device_capacitance_checks': self.device_capacitance_checks, 

202 'peak_memory': {'process_bytes': memory.process, 'largest_subprocess_bytes': memory.largest_subprocess}, 

203 'files': [written.to_json() for written in self.files], 

204 'warnings': self.warnings, 

205 'errors': self.errors, 

206 }

208 def write(self, announce: bool = True) -> bool: 

209 """ 

210 Writes the result file to its paths (see paths), each as a whole, and logs the first failure of each path 

211 as an error, but the run goes on (e.g. a reader on Windows may keep it open for a moment) 

212 

213 :param announce: whether to log the first write to each path (e.g. not before the log files are open) 

214 :return: whether it could write each of them 

215 """ 

216 text = json.dumps(self.to_json(), indent=2, ensure_ascii=False) + '\n' 

217 all_written = True 

218 for path in self.paths(): 

219 temporary_path = f"{path}.tmp" 

220 try: 

221 os.makedirs(os.path.dirname(path), exist_ok=True) 

222 with open(temporary_path, 'w', encoding='utf-8') as f: 

223 f.write(text) 

224 os.replace(temporary_path, path) 

225 except OSError as e: 

226 all_written = False 

227 if path not in self._failed_paths: 

228 self._failed_paths.add(path) 

229 error(f"Can't write the result file to: {path}: {e}") 

230 continue 

231 if announce and path not in self._announced_paths: 

232 self._announced_paths.add(path) 

233 info(f"Wrote result file to: {path}") 

234 return all_written

236 @contextlib.contextmanager 

237 def recording(self) -> Iterator[RunResult]: 

238 """ 

239 Collects the warnings and errors logged within, and writes the result file in the end, 

240 with the exit code of the run, also if it fails 

241 """ 

242 handler = _LoggedProblems(self) 

243 register_additional_handler(handler) 

244 try: 

245 yield self 

246 self.exit_code = int(ExitCode.OK) 

247 except SystemExit as e: 

248 # NOTE: as Python exits with it, e.g. 1 for a message 

249 self.exit_code = 0 if e.code is None else int(e.code) if isinstance(e.code, int) else 1 

250 raise 

251 except Exception as e: 

252 self.exit_code = 1 # NOTE: as Python exits with it, after the traceback 

253 self.errors.append(f"{type(e).__name__}: {e}") 

254 raise 

255 finally: 

256 deregister_additional_handler(handler) 

257 if self.exit_code == ExitCode.OK: 

258 self.stage = Stage.DONE 

259 self.write()
