36class ExportError(RuntimeError): 

37 """The scene could not be written out for the requested target."""

40class ExporterUnavailable(ExportError): 

41 """The backend for a target is not loadable in this installation."""

44class SceneFeature(Enum): 

45 """Scene contracts an exporter must implement before receiving them.""" 

46 

47 GROUND_CUTOUTS = 'ground-cutouts' 

48 """Explicit polygon openings through the ground slab and its coatings.""" 

49 

50 ELECTRICAL_IDENTITY = 'electrical-identity' 

51 """Ground reference net and FLOATING identity independent of output labels."""

54class PEX25DSceneExporter(Protocol): 

55 """Write solver inputs from a resolved ``PEX25DScene`` without running it. 

56 

57 The caller owns the scene; exporters must not mutate it. Implementations 

58 validate their own options and reject unsupported features with ExportError. 

59 NotImplementedError marks what an exporter does not implement yet. 

60 Geometry libraries and other optional dependencies should be imported only 

61 when needed, so metadata discovery remains usable without them. 

62 """ 

63 

64 scene_features: FrozenSet[SceneFeature] = frozenset() 

65 """Required feature contracts understood by this exporter. 

66 

67 An exporter without this declaration supports only scenes requiring none 

68 of these additions. A declaration promises actual geometry and ownership 

69 support; accepting an unfamiliar protobuf field is insufficient. 

70 """ 

71

78 """Return generated paths, primary input first, as in ``pex25d.export``.

88def require_scene_features(exporter: PEX25DSceneExporter, scene: Any, target: str): 

89 """Fail before writing if a consumer does not implement required scene semantics.""" 

90 from ...pex25d.electrical import solver_body_names 

91 required = set() 

92 if scene.HasField('ground_plane'): 

93 ground = scene.ground_plane 

94 if ground.cutouts: 

95 required.add(SceneFeature.GROUND_CUTOUTS) 

96 if ground.net and ground.net != ground.name: 

97 required.add(SceneFeature.ELECTRICAL_IDENTITY) 

98 if any(body.name != name for body, name in solver_body_names(scene).items()): 

99 required.add(SceneFeature.ELECTRICAL_IDENTITY) 

100 if not required: 

101 return 

102 declared = getattr(exporter, 'scene_features', frozenset()) 

103 # Older Protocol subclasses can inherit placeholders. A stub, None, 

104 # Ellipsis, or a similarly spelled string never promises implementation. 

105 features = declared if isinstance(declared, (set, frozenset)) else frozenset() 

106 missing = required - features 

107 if missing: 

108 names = ', '.join(sorted(feature.value for feature in missing)) 

109 raise ExportError(f"Exporter '{target}' does not support required scene features: " 

110 f"{names}; update the exporter before using this scene")
