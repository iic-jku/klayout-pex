25"""Version 1 PEX25D scene exporter protocol and its errors.""" 

26 

27from __future__ import annotations 

28 

29from typing import TYPE_CHECKING, Any, List, Mapping, Optional, Protocol 

30 

31if TYPE_CHECKING: 

32 from klayout_pex_protobuf.kpex.pex25d.pex25d_scene_pb2 import PEX25DScene 

33 

34 

35class ExportError(RuntimeError): 

36 """The scene could not be written out for the requested target.""" 

37 

38 

39class ExporterUnavailable(ExportError): 

40 """The backend for a target is not loadable in this installation.""" 

41 

42 

43class PEX25DSceneExporter(Protocol): 

44 """Write solver inputs from a resolved ``PEX25DScene`` without running it. 

45 

46 The caller owns the scene; exporters must not mutate it. Implementations 

47 validate their own options and reject unsupported features with ExportError. 

48 NotImplementedError marks what an exporter does not implement yet. 

49 Geometry libraries and other optional dependencies should be imported only 

50 when needed, so metadata discovery remains usable without them. 

51 """ 

52 

53 def export(self, 

54 scene: PEX25DScene, 

55 *, 

56 output_dir_path: str, 

57 prefix: str = '', 

58 options: Optional[Mapping[str, Any]] = None) -> List[str]: 

59 """Return generated paths, primary input first, as in ``pex25d.export``. 

60 

61 ``scene`` is a resolved PEX25DScene protobuf message. ``options`` contains 

62 backend-specific settings and must not be mutated. An empty ``prefix`` 

63 asks for the backend's default. Files are written under output_dir_path; 

64 returned paths must be usable by the caller without changing directory. 

65 """
