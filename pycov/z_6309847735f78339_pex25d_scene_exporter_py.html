36class ExportError(RuntimeError): 

37 """The scene could not be written out for the requested target."""

40class ExporterUnavailable(ExportError): 

41 """The backend for a target is not loadable in this installation."""

44class SceneCapability(Enum): 

45 """Scene contracts an exporter must implement before receiving them.""" 

46 

47 CAPACITANCE_ACCOUNTING = 'capacitance-accounting' 

48 """Typed exclusion masks, charge groups and result filtering.""" 

49 

50 GROUND_CUTOUTS = 'ground-cutouts' 

51 """Explicit polygon openings through the ground slab and its coatings.""" 

52 

53 ELECTRICAL_IDENTITY = 'electrical-identity' 

54 """Ground reference net and FLOATING identity independent of output labels."""

57class PEX25DSceneExporter(Protocol): 

58 """Write solver inputs from a resolved ``PEX25DScene`` without running it. 

59 

60 The caller owns the scene; exporters must not mutate it. Implementations 

61 validate their own options and reject unsupported capabilities with ExportError. 

62 NotImplementedError marks what an exporter does not implement yet. 

63 Geometry libraries and other optional dependencies should be imported only 

64 when needed, so metadata discovery remains usable without them. 

65 """ 

66 

67 @classmethod 

68 def supported_capabilities(cls) -> FrozenSet[SceneCapability]: 

69 """Return the scene contracts implemented by this exporter class. 

70 

71 Querying support must not construct an exporter, import optional geometry 

72 libraries, or write output. Derived exporters may extend this immutable 

73 set with ``super().supported_capabilities()``. The default promises no 

74 additional capabilities; accepting protobuf fields alone is not support. 

75 """ 

76 return frozenset()

94def require_scene_capabilities(exporter: PEX25DSceneExporter, scene: Any, target: str): 

95 """Fail before writing if a consumer does not implement required scene semantics.""" 

96 from ...pex25d.electrical import solver_body_names 

97 required: Set[SceneCapability] = set() 

98 declared_capabilities = getattr(scene, 'required_capabilities', ()) 

99 has_policy = getattr(scene, 'capacitance_accounting', None) is not None and scene.HasField('capacitance_accounting') 

100 has_parts = any(getattr(region, 'accounting_part_id', '') or getattr(region, 'accounting_regions', ()) 

101 for conductor in scene.conductors for region in getattr(conductor, 'regions', ())) 

102 if declared_capabilities or has_policy or has_parts: 

103 from ...pex25d.capability_validation import validate_required_capabilities 

104 from ...pex25d.diagnostics import DiagnosticsReport 

105 from ...pex25d.protobuf import pex25d_capabilities_pb2 

106 report = DiagnosticsReport() 

107 if not validate_required_capabilities(scene, report): 

108 raise ExportError('Unsupported or missing required scene capability: ' + 

109 '; '.join(d.message for d in report.diagnostics)) 

110 accounting = pex25d_capabilities_pb2().REQUIRED_CAPABILITY_CAPACITANCE_ACCOUNTING 

111 if accounting in declared_capabilities: 

112 required.add(SceneCapability.CAPACITANCE_ACCOUNTING) 

113 if scene.HasField('ground_plane'): 

114 ground = scene.ground_plane 

115 if ground.cutouts: 

116 required.add(SceneCapability.GROUND_CUTOUTS) 

117 if ground.net and ground.net != ground.name: 

118 required.add(SceneCapability.ELECTRICAL_IDENTITY) 

119 if any(body.name != name for body, name in solver_body_names(scene).items()): 

120 required.add(SceneCapability.ELECTRICAL_IDENTITY) 

121 if not required: 

122 return 

123 declaration = getattr(exporter, 'supported_capabilities', None) 

124 # Structural implementations without the method support ordinary scenes only. 

125 if declaration is None: 

126 supported = frozenset() 

127 else: 

128 if not callable(declaration): 

129 raise ExportError(f"Exporter '{target}' must declare supported_capabilities() as a class method") 

130 supported = declaration() 

131 if not isinstance(supported, frozenset) or any( 

132 not isinstance(capability, SceneCapability) for capability in supported): 

133 raise ExportError(f"Exporter '{target}' supported_capabilities() must return " 

134 "a frozenset of SceneCapability members") 

135 missing = required - supported 

136 if missing: 

137 names = ', '.join(sorted(capability.value for capability in missing)) 

138 raise ExportError(f"Exporter '{target}' does not support required scene capabilities: " 

139 f"{names}; update the exporter before using this scene")
