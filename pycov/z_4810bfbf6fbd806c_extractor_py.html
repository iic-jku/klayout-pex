80def devices_of_cut_cells(exclusions: Exclusions, 

81 pex_context: KLayoutExtractionContext, 

82 tech_info: TechInfo) -> Dict[int, int]: 

83 """ 

84 :return: the devices within the cut cells by their IDs, with the indexes of their blocks: those that a cell draws, 

85 by the layers that LVS makes them from (see TechInfo.device_gds_pairs), at their terminals, but those 

86 on the substrate, and on a tap, which goes on beyond the cell, unless the device is a tap 

87 (see CutCellInstance.draws_device) 

88 

89 NOTE: by geometry, as the LVS netlist is flat 

90 """ 

91 cut_block_indexes = {index for index, block in enumerate(exclusions.blocks) if block.kind == BlockKind.CUT} 

92 if not cut_block_indexes: 

93 return {} 

94 

95 lvsdb = pex_context.lvsdb 

96 substrate_lvs_layer_names = set(tech_info.tech.substrate.lvs_layer_names) 

97 gds_pairs_by_class: Dict[str, Optional[Set[GDSPair]]] = {} 

98 blocks: Dict[int, int] = {} 

99 for device in pex_context.top_circuit.each_device(): 

100 class_name = device.device_class().name 

101 if class_name not in gds_pairs_by_class: 

102 gds_pairs_by_class[class_name] = tech_info.device_gds_pairs(class_name) 

103 gds_pairs = gds_pairs_by_class[class_name] 

104 if gds_pairs is None: 

105 continue 

106 

107 terminals: List[kdb.Region] = [] 

108 on_taps: List[kdb.Region] = [] 

109 is_a_tap = True 

110 bbox = kdb.Box() 

111 for terminal_definition in device.device_class().terminal_definitions(): 

112 for reference in pex_context.terminal_refs[(device.id(), terminal_definition.id())]: 

113 for index, shapes in pex_context.shapes_of_terminal(device, reference).items(): 

114 lvs_layer_name = lvsdb.layer_name(index) 

115 if shapes.is_empty() or lvs_layer_name in substrate_lvs_layer_names: 

116 continue 

117 if is_on_tap(lvs_layer_name, tech_info): 

118 on_taps.append(shapes) 

119 continue 

120 terminals.append(shapes) 

121 bbox += shapes.bbox() 

122 is_a_tap &= is_on_well(lvs_layer_name, tech_info) 

123 if is_a_tap: 

124 terminals += on_taps 

125 for shapes in on_taps: 

126 bbox += shapes.bbox() 

127 

128 block_index = next((index for index in exclusions.block_indexes_touching(bbox) 

129 if index in cut_block_indexes 

130 and exclusions.blocks[index].cell_instance.draws_device(gds_pairs, terminals)), None) 

131 if block_index is not None: 

132 blocks[device.id()] = block_index 

133 

134 unknown_classes = sorted(name for name, gds_pairs in gds_pairs_by_class.items() if gds_pairs is None) 

135 if unknown_classes: 

136 warning(f"The tech info doesn't tell what LVS makes these device classes from, so the netlist keeps " 

137 f"their devices, also of the cut cells: {', '.join(unknown_classes)}") 

138 return blocks

141def add_filled_diffusion_to_cut_cells(exclusions: Exclusions, 

142 pex_context: KLayoutExtractionContext, 

143 tech_info: TechInfo): 

144 """ 

145 The cut cells stand for what LVS computes of their devices where the layout draws no diffusion, 

146 e.g. the drain of an LDMOS, across the gap to its gate 

147 """ 

148 diffusion_layer_names = set(tech_info.process_diffusion_layer_names) 

149 diffusion_gds_pairs = CutLayers.of_tech(tech_info).diffusion_gds_pairs 

150 for device in pex_context.top_circuit.each_device(): 

151 block_index = exclusions.removed_device_blocks.get(device.id(), None) 

152 if block_index is None: 

153 continue 

154 instance = exclusions.blocks[block_index].cell_instance 

155 for terminal_definition in device.device_class().terminal_definitions(): 

156 for reference in pex_context.terminal_refs[(device.id(), terminal_definition.id())]: 

157 for index, shapes in pex_context.shapes_of_terminal(device, reference).items(): 

158 lvs_layer_name = pex_context.lvsdb.layer_name(index) 

159 computed_layer_info = tech_info.computed_layer_info_by_name.get(lvs_layer_name, None) 

160 layer_name = computed_layer_info.original_layer_name if computed_layer_info is not None else None 

161 # NOTE: not the box of a ring (e.g. the collector of a Schottky diode), which is no gap of a device 

162 if layer_name in diffusion_layer_names and not is_on_tap(lvs_layer_name, tech_info): 

163 filled = shapes - instance.shapes_of_layout(diffusion_gds_pairs, shapes.bbox()) 

164 if not filled.is_empty(): 

165 exclusions.add_to_cut_cell(block_index, layer_name, filled)

168def prepare_exclusions(exclusions: Exclusions, 

169 pex_context: KLayoutExtractionContext, 

170 tech_info: TechInfo): 

171 """ 

172 Finds the nets within the exclusion blocks that nothing connects to the netlist, 

173 and warns of what the extraction leaves out with the markers beyond what they stand for 

174 """ 

175 if not exclusions.blocks: 

176 return 

177 

178 exclusions.removed_device_blocks = devices_of_cut_cells(exclusions, pex_context, tech_info) 

179 if exclusions.removed_device_blocks: 

180 info(f"The netlist leaves out the {len(exclusions.removed_device_blocks)} devices of the cut cells") 

181 add_filled_diffusion_to_cut_cells(exclusions, pex_context, tech_info) 

182 

183 K = tech_pb2.ComputedLayerInfo.Kind 

184 layers_within: Dict[NetName, Set[LayerName]] = defaultdict(set) 

185 layers_within_markers: Dict[NetName, Set[LayerName]] = defaultdict(set) 

186 nets_outside: Set[NetName] = set() 

187 nets_outside_markers: Set[NetName] = set() 

188 # NOTE: not the wells and the substrate below the markers, which are in the models of what they stand for 

189 for gds_pair in tech_info.process_conductor_gds_pairs: 

190 extracted_layer = pex_context.extracted_layers.get(gds_pair, None) 

191 if extracted_layer is None: 

192 continue 

193 layer_name = tech_info.canonical_layer_name_by_gds_pair[gds_pair] 

194 mask = exclusions.mask(layer_name) 

195 marker_mask = exclusions.mask(layer_name, kind=BlockKind.MARKER) 

196 for source_layer in extracted_layer.source_layers: 

197 kind = tech_info.computed_layer_info_by_name[source_layer.lvs_layer_name].kind 

198 if kind in (K.KIND_PIN, K.KIND_LABEL): 

199 continue 

200 shapes_by_net = pex_context.shapes_by_net_name_by_lvs_layer[source_layer.lvs_layer_name] 

201 for net_name, shapes in shapes_by_net.items(): 

202 for region, within, outside in ((mask, layers_within, nets_outside), 

203 (marker_mask, layers_within_markers, nets_outside_markers)): 

204 box = shapes.bbox() 

205 if not box.touches(region.bbox()): 

206 outside.add(net_name) 

207 continue 

208 near = exclusions.regions.near(region, box) 

209 if not (shapes & near).is_empty(): 

210 within[net_name].add(layer_name) 

211 if not (shapes - near).is_empty(): 

212 outside.add(net_name) 

213 

214 circuit = pex_context.top_circuit 

215 blackbox_devices = pex_context.blackbox_devices 

216 device_models = tech_info.device_models 

217 

218 def is_whiteboxed(device: kdb.Device) -> bool: 

219 return not blackbox_devices and RCX25NetlistExpander.is_whiteboxed(device.device_class(), device_models) 

220 

221 def is_removed(device: kdb.Device) -> bool: 

222 return device.id() in exclusions.removed_device_blocks or is_whiteboxed(device) 

223 

224 def is_connected(net_name: NetName) -> bool: 

225 net = circuit.net_by_name(net_name) 

226 return net is None or net.pin_count() + net.subcircuit_pin_count() > 0 or \ 

227 any(not is_removed(terminal.device()) for terminal in net.each_terminal()) 

228 

229 exclusions.floating_net_names = {net_name for net_name in layers_within 

230 if net_name not in nets_outside and not is_connected(net_name)} 

231 

232 nets_across = sorted(net_name for net_name in layers_within_markers if net_name in nets_outside_markers) 

233 if nets_across: 

234 listed_count = 20 

235 warning("These nets are partly within exclusion markers, so the extraction leaves out their parasitics " 

236 "there, for the model of what the markers stand for (a wire across a marker loses them too):\n" + 

237 '\n'.join(f" - {net_name}: {', '.join(sorted(layers_within_markers[net_name]))}" 

238 for net_name in nets_across[:listed_count]) + 

239 (f"\n - and {len(nets_across) - listed_count} more" if len(nets_across) > listed_count else '')) 

240 if exclusions.floating_net_names: 

241 info(f"Nothing connects these nets within exclusion markers to the netlist, " 

242 f"so their capacitances to the rest of the layout go to the substrate: " 

243 f"{listed_names(sorted(exclusions.floating_net_names))}") 

244 

245 substrate_lvs_layer_names = set(tech_info.tech.substrate.lvs_layer_names)

247 def is_plate_within(block: ExcludedBlock, lvs_layer_name: LVSLayerName, shapes: kdb.Region) -> bool: 

248 computed_layer_info = tech_info.computed_layer_info_by_name.get(lvs_layer_name, None) 

249 region = block.region if computed_layer_info is None \ 

250 else block.region_of(computed_layer_info.original_layer_name) 

251 return region is not None and not (shapes & exclusions.regions.near(region, shapes.bbox())).is_empty()

253 def has_plates_within_a_block(device: kdb.Device) -> bool: 

254 """ 

255 NOTE: all of them, as what a capacitance is left out by has both of its plates within one block 

256 """ 

257 plates: List[List[Tuple[LVSLayerName, kdb.Region]]] = [] 

258 box = kdb.Box() 

259 for terminal in device.device_class().terminal_definitions(): 

260 plate: List[Tuple[LVSLayerName, kdb.Region]] = [] 

261 for reference in pex_context.terminal_refs[(device.id(), terminal.id())]: 

262 for index, shapes in pex_context.shapes_of_terminal(device, reference).items(): 

263 lvs_layer_name = pex_context.lvsdb.layer_name(index) 

264 if not shapes.is_empty() and lvs_layer_name not in substrate_lvs_layer_names: 

265 plate.append((lvs_layer_name, shapes)) 

266 box += shapes.bbox() 

267 if plate: 

268 plates.append(plate) 

269 return any(block.capacitance_pair is None and 

270 all(any(is_plate_within(block, *part) for part in plate) for plate in plates) 

271 for block in exclusions.blocks_touching(box)) 

272 

273 lost_devices = [d.expanded_name() for d in circuit.each_device() 

274 if is_whiteboxed(d) and d.id() not in exclusions.removed_device_blocks 

275 and has_plates_within_a_block(d)] 

276 if lost_devices: 

277 warning(f"The netlist loses these devices within exclusion markers, as white-box mode (--blackbox n) " 

278 f"removes them, and the capacitances of their plates are left out, use --blackbox y to keep them: " 

279 f"{listed_names(sorted(lost_devices))}")

335 def substrate_region(self) -> kdb.Region: 

336 """ 

337 The substrate below the layout (and its halo), the bottom plate of the capacitances to the substrate, 

338 split into its wells, of the nets of the wells (e.g. VDD), and the rest, of the substrate (VSUBS) 

339 

340 NOTE: with --blackbox, its shapes are plates of the capacitor devices with a terminal on the substrate 

341 (e.g. a VPP cap), see DeviceCapacitorPlates 

342 """ 

343 dbu = self.pex_context.dbu 

344 side_halo_um = self.tech_info.tech.process_parasitics.side_halo 

345 rest = kdb.Region(self.pex_context.top_cell_bbox().enlarged(side_halo_um / dbu)) # e.g. 8 µm halo 

346 

347 # NOTE: the channels of the transistors aren't substrate the gate poly couples to, 

348 # their capacitances are the transistor models' (#290) 

349 rest -= self.pex_context.mos_channels 

350 

351 substrate_region = kdb.Region() 

352 substrate_region.enable_properties() 

353 

354 plate_properties = {DEVICE_CAPACITOR_PLATE_PROPERTY: True} if self.pex_context.blackbox_devices else {} 

355 

356 def insert(region: kdb.Region, net_name: NetName): 

357 for polygon in region.each(): 

358 substrate_region.insert(kdb.PolygonWithProperties(polygon.downcast(), 

359 {'net': net_name, **plate_properties})) 

360 

361 # NOTE: where wells overlap (e.g. an nwell within a deep nwell), the first one is at the surface 

362 for well_layer_name in self.tech_info.tech.substrate.well_lvs_layer_names: 

363 gds_pair = self.gds_pair(layer_name=well_layer_name) 

364 if gds_pair is None: 

365 continue 

366 # NOTE: the well's LVS layer only, not the others of its GDS pair (e.g. the pwell around an isolated pwell) 

367 wells = self.pex_context.shapes_of_layer(gds_pair=gds_pair, lvs_layer_names=[well_layer_name]) 

368 if wells is None: 

369 continue 

370 for well in wells.merged().each(): # NOTE: merged per net 

371 well_region = kdb.Region(well.downcast()) & rest 

372 rest -= well_region 

373 insert(well_region, well.property('net')) 

374 

375 insert(rest, self.tech_info.internal_substrate_layer_name) 

376 return substrate_region

378 def extract(self) -> ExtractionResults: 

379 extraction_results = ExtractionResults() 

380 

381 # TODO: for now, we always flatten and have only 1 cell 

382 cell_name = self.pex_context.annotated_top_cell.name 

383 extraction_report = ExtractionReporter(cell_name=cell_name, 

384 dbu=self.pex_context.dbu, 

385 report_caps=self.report_caps) 

386 # NOTE: in RC mode, the capacitances go onto the resistor network, by their geometry 

387 cell_extraction_results = CellExtractionResults( 

388 cell_name=cell_name, 

389 keep_capacitance_geometry=self.pex_mode == PEXMode.RC, 

390 substrate_layer_name=self.tech_info.internal_substrate_layer_name, 

391 substrate_well_layer_names=self.tech_info.substrate_well_layer_names, 

392 exclusions=self.exclusions 

393 ) 

394 

395 # Explicitly log the stacktrace here, because otherwise Exceptions  

396 # raised in the callbacks of *NeighborhoodVisitors can cause RuntimeErrors 

397 # that are not traceable beyond the Region.complex_op() calls 

398 try: 

399 self.extract_cell(results=cell_extraction_results, 

400 report=extraction_report) 

401 except RuntimeError as e: 

402 import traceback 

403 print(f"Caught a RuntimeError: {e}") 

404 traceback.print_exc() 

405 raise 

406 

407 extraction_results.cell_extraction_results[cell_name] = cell_extraction_results 

408 

409 extraction_report.save(self.report_path) 

410 

411 return extraction_results

413 def extract_cell(self, 

414 results: CellExtractionResults, 

415 report: ExtractionReporter): 

416 netlist: kdb.Netlist = self.pex_context.lvsdb.netlist() 

417 dbu = self.pex_context.dbu 

418 

419 if self.exclusions is not None and not self.pex_mode.need_resistance(): 

420 warn_of_ports_without_nodes(self.exclusions, f"--mode {self.pex_mode}") 

421 # ------------------------------------------------------------------------ 

422 

423 layer_regions_by_name: Dict[LayerName, kdb.Region] = defaultdict(kdb.Region) 

424 

425 all_region = kdb.Region() 

426 all_region.enable_properties() 

427 

428 layer_regions_by_name[self.tech_info.internal_substrate_layer_name] = self.substrate_region() 

429 

430 # NOTE: the diffusion (the source/drain of the transistors) is above the substrate, which it shields, 

431 # and below the metal layers, of whose capacitances it's the bottom plate 

432 for diffusion_layer in self.tech_info.process_diffusion_layers: 

433 diffusion_shapes = self.shapes_of_layer(diffusion_layer.name) 

434 if diffusion_shapes is not None: 

435 diffusion_shapes.enable_properties() 

436 gds_pair = self.gds_pair(diffusion_layer.name) 

437 canonical_layer_name = self.tech_info.canonical_layer_name_by_gds_pair[gds_pair] 

438 layer_regions_by_name[canonical_layer_name] += diffusion_shapes 

439 layer_regions_by_name[canonical_layer_name].enable_properties() 

440 

441 via_name_below_layer_name: Dict[LayerName, Optional[LayerName]] = {} 

442 via_name_above_layer_name: Dict[LayerName, Optional[LayerName]] = {} 

443 via_regions_by_via_name: Dict[LayerName, kdb.Region] = defaultdict(kdb.Region) 

444 

445 previous_via_name: Optional[str] = None 

446 

447 for metal_layer in self.tech_info.process_metal_layers: 

448 layer_name = metal_layer.name 

449 gds_pair = self.gds_pair(layer_name) 

450 canonical_layer_name = self.tech_info.canonical_layer_name_by_gds_pair[gds_pair] 

451 

452 # NOTE: with --blackbox, the plates of the capacitor devices are marked (see DeviceCapacitorPlates) 

453 all_layer_shapes = self.shapes_of_layer(layer_name, 

454 mark_device_capacitor_plates=self.pex_context.blackbox_devices) 

455 if all_layer_shapes is not None: 

456 all_layer_shapes.enable_properties() 

457 

458 layer_regions_by_name[canonical_layer_name] += all_layer_shapes 

459 layer_regions_by_name[canonical_layer_name].enable_properties() 

460 all_region += all_layer_shapes 

461 

462 if metal_layer.metal_layer.HasField('contact_above'): 

463 contact = metal_layer.metal_layer.contact_above 

464 

465 via_regions = self.shapes_of_layer(contact.name) 

466 if via_regions is not None: 

467 via_regions.enable_properties() 

468 via_regions_by_via_name[contact.name] += via_regions 

469 via_name_above_layer_name[canonical_layer_name] = contact.name 

470 via_name_below_layer_name[canonical_layer_name] = previous_via_name 

471 

472 previous_via_name = contact.name 

473 else: 

474 previous_via_name = None 

475 

476 all_layer_names = list(layer_regions_by_name.keys()) 

477
