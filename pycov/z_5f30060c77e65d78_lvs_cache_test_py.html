67class FakeKLayoutLVS: 

68 """ 

69 Stands in for subprocess.Popen, emulating a successful KLayout LVS run 

70 """ 

71 def __init__(self): 

72 self.call_count = 0 

73 

74 def __call__(self, args: List[str], **kwargs): 

75 self.call_count += 1 

76 report_path = next(a.removeprefix('report=') for a in args if a.startswith('report=')) 

77 write_file(report_path, '#%lvsdb-klayout\n') 

78 proc = mock.Mock() 

79 proc.stdout = io.StringIO("") 

80 proc.returncode = 0 

81 return proc

84class TempDirTestCase(unittest.TestCase): 

85 def setUp(self): 

86 self.tmp_dir = tempfile.mkdtemp(prefix="lvs_cache_") 

87 self.addCleanup(shutil.rmtree, self.tmp_dir, ignore_errors=True) 

88 

89 def make_deck(self, dir_name: str = 'deck') -> str: 

90 deck_dir_path = os.path.join(self.tmp_dir, dir_name) 

91 write_file(os.path.join(deck_dir_path, 'sg13g2.lvs'), '# %include rule_decks/globals.lvs\n') 

92 write_file(os.path.join(deck_dir_path, 'rule_decks', 'globals.lvs'), 'RF_TO_BASE_MOS_MODEL = {}\n') 

93 return os.path.join(deck_dir_path, 'sg13g2.lvs') 

94 

95 def fingerprint(self, **overrides) -> LVSInputFingerprint: 

96 kwargs = dict(pdk='ihp-sg13g2', 

97 klayout_version='0.30.12', 

98 lvs_script_path=os.path.join(self.tmp_dir, 'deck', 'sg13g2.lvs'), 

99 script_parameters={'run_mode': 'deep', 'no_simplify': 'true'}, 

100 gds_path=os.path.join(self.tmp_dir, 'cell.gds'), 

101 cell_name='cell', 

102 schematic_path=os.path.join(self.tmp_dir, 'cell.spice')) 

103 kwargs.update(overrides) 

104 return lvs_input_fingerprint(**kwargs)

130 def test_deck_follows_symbolic_links(self): 

131 # e.g. the ihp-sg13cmos5l rule decks, which link to those of ihp-sg13g2

133 write_file(shared_path, 'v1') 

134 link_path = os.path.join(os.path.dirname(self.lvs_script_path), 'rule_decks', 'shared.lvs') 

135 os.symlink(shared_path, link_path) 

136 os.symlink(os.path.join(self.tmp_dir, 'missing.lvs'), 

137 os.path.join(os.path.dirname(self.lvs_script_path), 'rule_decks', 'dangling.lvs')) 

138 before = deck_fingerprint(self.lvs_script_path) 

139 write_file(shared_path, 'v2') 

140 after = deck_fingerprint(self.lvs_script_path) 

141 self.assertNotIn('rule_decks/dangling.lvs', before.sha256_by_relative_path) 

142 self.assertNotEqual(before.sha256_by_relative_path['rule_decks/shared.lvs'], 

143 after.sha256_by_relative_path['rule_decks/shared.lvs'])

222 def setUp(self): 

223 super().setUp() 

224 

225 self.output_dir_path = os.path.join(self.tmp_dir, 'output', f"{CELL_NAME}__{CELL_NAME}") 

226 os.makedirs(self.output_dir_path) 

227 self.lvsdb_path = os.path.join(self.output_dir_path, f"{CELL_NAME}.lvsdb.gz") 

228 

229 self.gds_path = os.path.join(self.tmp_dir, f"{CELL_NAME}.gds") 

230 write_file(self.gds_path, 'layout') 

231 schematic_path = os.path.join(self.output_dir_path, f"{CELL_NAME}_dummy_schematic.spice") 

232 write_file(schematic_path, f".subckt {CELL_NAME} VDD VSS\n.ends\n.end\n") 

233 

234 self.args = argparse.Namespace( 

235 input_mode=InputMode.GDS, 

236 output_dir_path=self.output_dir_path, 

237 cache_dir_path=os.path.join(self.tmp_dir, 'output', '.kpex_cache'), 

238 cache_lvs=True, 

239 pdk='ihp-sg13g2', 

240 gds_path=self.gds_path, 

241 effective_gds_path=self.gds_path, 

242 effective_schematic_path=schematic_path, 

243 effective_cell_name=CELL_NAME, 

244 klayout_exe_path='klayout', 

245 lvs_script_path=self.make_deck(), 

246 klayout_lvs_verbose=False, 

247 ) 

248 self.cache_entry = LVSCacheEntry( 

249 dir_path=os.path.join(self.args.cache_dir_path, self.args.pdk, 

250 os.path.splitroot(os.path.abspath(self.gds_path))[-1]), 

251 cell_name=CELL_NAME 

252 ) 

253 

254 patcher = mock.patch.object(Tool, 'detect_version', return_value=Version('0.30.12')) 

255 patcher.start() 

256 self.addCleanup(patcher.stop) 

257 

258 def make_cache_hit(self): 

259 self.cache_entry.store(lvsdb_path=CACHED_LVSDB_PATH, 

260 fingerprint=KpexCLI.lvs_input_fingerprint(self.args)) 

261 

262 def create_lvsdb(self) -> Tuple[Any, FakeKLayoutLVS, mock.Mock]: 

263 fake_klayout = FakeKLayoutLVS() 

264 with mock.patch('klayout_pex.klayout.lvs_runner.subprocess.Popen', fake_klayout), \ 

265 mock.patch('klayout_pex.kpex_cli.info') as info_mock: 

266 lvsdb = KpexCLI().create_lvsdb(self.args) 

267 return lvsdb, fake_klayout, info_mock

269 def assert_cache_miss(self, reason: str, fake_klayout: FakeKLayoutLVS, info_mock: mock.Mock): 

270 self.assertEqual(1, fake_klayout.call_count) 

271 info_mock.assert_any_call(f"Cache miss: {reason}") 

272 # the new LVS database is cached, together with its fingerprint 

273 self.assertTrue(filecmp.cmp(self.lvsdb_path, self.cache_entry.lvsdb_path, shallow=False)) 

274 self.assertEqual(KpexCLI.lvs_input_fingerprint(self.args), 

275 read_fingerprint(self.cache_entry.fingerprint_path))

277 def test_cache_hit_reads_the_cached_lvsdb(self):
