25"""The exporter ``gmsh``: the mesh on its own, for any FEM tool or for looking at.""" 

26 

27from __future__ import annotations 

28 

29from typing import TYPE_CHECKING, Any, ClassVar, List, Mapping, Optional 

30 

31from klayout_pex.plugin_api.v1 import PEX25DSceneExporter 

32 

33from .mesh import write_mesh 

34from .options import GmshMeshOptions, parse_options 

35 

36if TYPE_CHECKING: 

37 from klayout_pex_protobuf.kpex.pex25d.pex25d_scene_pb2 import PEX25DScene 

38 

39 

40class GmshSceneExporter(PEX25DSceneExporter): 

41 """Write a gmsh volume mesh of a resolved PEX25D scene.""" 

42 

43 name: ClassVar[str] = 'gmsh' 

44 default_prefix: ClassVar[str] = '' 

45 

46 def export(self, 

47 scene: PEX25DScene, 

48 *, 

49 output_dir_path: str, 

50 prefix: str = '', 

51 options: Optional[Mapping[str, Any]] = None) -> List[str]: 

52 """Return the mesh first, then the JSON naming its physical groups.""" 

53 settings = parse_options(GmshMeshOptions, options, self.name) 

54 result = write_mesh(scene, settings, output_dir_path, prefix or self.default_prefix) 

55 return result.paths
