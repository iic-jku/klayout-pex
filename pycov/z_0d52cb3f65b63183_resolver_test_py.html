38def resolved(text: str = MINIMAL) -> Any: 

39 return resolve(read(text)) 

40 

41 

42def resolve_codes(text: str) -> list: 

43 report = DiagnosticsReport() 

44 try: 

45 resolve(read(text), report=report) 

46 except ResolveError: 

47 pass 

48 return codes(report)

51def layer(scene: Any, name: str) -> Any: 

52 return next(l for l in scene.layers if l.name == name) 

53 

54 

55def dielectric(scene: Any, name: str) -> Any: 

56 return next(d for d in scene.dielectrics if d.name == name)

64 def test_a_via_takes_its_extent_from_connects(self): 

65 scene = resolved() 

66 via = layer(scene, 'via1') 

67 assert (via.zlow, via.zhigh) == (14000, 20000) # met1 top → met2 bottom 

68 assert (via.connects_below, via.connects_above) == ('met1', 'met2')

70 def test_a_via_may_land_on_the_ground_plane(self): 

71 scene = resolved(replacing('VIA via1 CONNECTS met1 met2', 

72 'VIA via1 CONNECTS subs met1')) 

73 assert (layer(scene, 'via1').zlow, layer(scene, 'via1').zhigh) == (-1000, 10000)

75 def test_a_simple_band_spans_bottom_to_bottom(self): 

76 # BETWEEN met1 met2 is the bottom of met1 to the bottom of met2, so the 

77 # band joins the neighbouring levels with no seam. 

78 band = dielectric(resolved(), 'ild') 

79 assert (band.zlow, band.zhigh) == (10000, 20000)

202 def test_domain_margin_expands_the_geometry_bounds(self): 

203 scene = resolved() 

204 origins = pex25d_scene_pb2().ResolvedDomain 

205 assert scene.domain.origin == origins.ORIGIN_DOMAIN_MARGIN 

206 bounds = scene.domain.geometry_bounds 

207 box = scene.domain.box 

208 assert box.lower_left.x == bounds.lower_left.x - 40000 

209 assert box.upper_right.z == bounds.upper_right.z + 20000 

210 

211 def test_geometry_bounds_ignore_laterally_unbounded_materials(self): 

212 # Only conductor shapes and conformal films with THICKNESS_ON_FIELD 0 

213 # are finite in XY. A simple band is not, and must not inflate this. 

214 bounds = resolved().domain.geometry_bounds 

215 assert bounds.lower_left.x == -800 # met1 at x=0, grown by lint 

216 assert bounds.upper_right.x == 30800 # conductor B at x=3.0, plus lint 

217 assert bounds.upper_right.z == 24000 # met2 top, not the ild band top 

218 

219 def test_no_domain_record_leaves_the_domain_unspecified(self): 

220 # The message is still present — it carries geometry_bounds for an 

221 # adapter placing its own boundary — but nothing in it is a domain. 

222 origins = pex25d_scene_pb2().ResolvedDomain 

223 scene = resolved(replacing('DOMAIN_MARGIN X 4.0 Y 4.0 Z 2.0\n', '')) 

224 assert scene.domain.origin == origins.ORIGIN_UNSPECIFIED 

225 assert not scene.domain.HasField('box') 

226 assert not scene.domain.HasField('applied_margin') 

227 assert scene.domain.geometry_bounds.upper_right.z == 24000
