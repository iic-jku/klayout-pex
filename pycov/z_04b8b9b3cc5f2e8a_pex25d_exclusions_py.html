42def scaled_region(region: kdb.Region, scale: kdb.ICplxTrans) -> kdb.Region: 

43 """Validate rounded bounds before a scaled mask reaches native indexing.""" 

44 box = region.bbox() 

45 if not box.empty(): 

46 from .native_coordinates import NATIVE_COORDINATE_MIN, NATIVE_COORDINATE_MAX 

47 for value in (box.left, box.bottom, box.right, box.top): 

48 scaled = value * scale.mag 

49 if not NATIVE_COORDINATE_MIN < scaled < NATIVE_COORDINATE_MAX: 

50 raise ValueError('Accounting geometry exceeds the native coordinate range') 

51 transformed = box.transformed(scale) 

52 for coordinate in (transformed.left, transformed.bottom, transformed.right, transformed.top): 

53 check_native_ground_coordinate(coordinate) 

54 return region.transformed(scale)

57def add_mask(target: Any, layer: str, geometry: kdb.Region): 

58 if geometry.is_empty(): 

59 return 

60 mask = target.masks.add(layer=layer) 

61 for polygon in geometry.each_merged(): 

62 item = mask.polygons.add() 

63 for point in polygon.each_point_hull(): 

64 item.outer.points.add(x=point.x, y=point.y) 

65 for hole_index in range(polygon.holes()): 

66 ring = item.holes.add() 

67 for point in polygon.each_point_hole(hole_index): 

68 ring.points.add(x=point.x, y=point.y)

71def attach_layout_accounting(file: Any, builder: Any, exclusions: Exclusions): 

72 """Preserve source-grid via ownership while serializing physical-grid masks.""" 

73 schema = pex25d_capacitance_accounting_pb2() 

74 file.required_capabilities.append(pex25d_capabilities_pb2().REQUIRED_CAPABILITY_CAPACITANCE_ACCOUNTING) 

75 file.format_version_suffix = ACCOUNTING_FORMAT_VERSION_SUFFIX 

76 policy = file.capacitance_accounting 

77 policy.SetInParent() 

78 net_names = {conductor.net for conductor in file.conductors} 

79 policy.excluded_substrate_nets.extend(sorted(exclusions.excluded_net_names & net_names)) 

80 grid = Fraction(file.units.grid_numerator, file.units.grid_denominator) 

81 scale = kdb.ICplxTrans(float(Fraction(str(builder.pex_context.dbu)) / grid)) 

82 tech = builder.tech_info 

83 

84 def canonical(name: str) -> str: 

85 pair = tech.gds_pair(name) 

86 return tech.canonical_layer_name_by_gds_pair.get(pair, name) 

87 

88 # Alias map is resolved here. Downstream exporters need no technology or LVS objects. 

89 aliases = {metal.name: canonical(metal.name) for metal in file.metals} 

90 aliases[file.ground_plane.name] = exclusions.substrate_layer_name 

91 contacts: Dict[str, Tuple[str, str]] = {} 

92 for layer in tech.process_metal_layers: 

93 contact = layer.metal_layer.contact_above 

94 if contact.name: 

95 contacts[contact.name] = (canonical(contact.layer_below or layer.name), 

96 canonical(contact.metal_above or layer.name)) 

97 for layer in tech.process_diffusion_layers: 

98 contact = layer.diffusion_layer.contact_above 

99 if contact.name: 

100 contacts[contact.name] = (canonical(layer.name), canonical(contact.metal_above or layer.name)) 

101 for layer in tech.process_nwell_layers: 

102 contact = layer.nwell_layer.contact_above 

103 if contact.name: 

104 contacts[contact.name] = (canonical(layer.name), canonical(contact.metal_above or layer.name)) 

105 via_geometry: Dict[str, kdb.Region] = {via.name: kdb.Region() for via in file.vias} 

106 for shape in file.shapes: 

107 if shape.layer in via_geometry: 

108 via_geometry[shape.layer] += shape_region(shape) 

109 

110 for index, block in enumerate(exclusions.blocks): 

111 region = policy.regions.add(name=f'block{index}') 

112 for profile, alias in aliases.items(): 

113 if profile == file.ground_plane.name: 

114 continue 

115 source = block.region_of(alias) 

116 if source is not None: 

117 add_mask(region, profile, scaled_region(source, scale)) 

118 if block.capacitance_pair is None: 

119 for via, physical in via_geometry.items(): 

120 if physical.is_empty() or via not in contacts: 

121 continue 

122 below, above = contacts[via] 

123 bottom, top = block.region_of(below), block.region_of(above) 

124 if bottom is None or top is None: 

125 continue 

126 # Cuts are unchanged by width bias. Classify before rescaling so 

127 # integer bbox-center rounding and the one-DBU margin stay fixed. 

128 source = physical.transformed(scale.inverted()) 

129 if not (source.transformed(scale) ^ physical).is_empty(): 

130 raise ValueError(f"Via '{via}' cannot be represented on its source grid") 

131 selected = polygons_by_center_inside(source, bottom & top) 

132 add_mask(region, via, scaled_region(selected, scale)) 

133 if not region.masks: 

134 del policy.regions[-1] 

135 continue 

136 if block.capacitance_pair is None: 

137 policy.exclusions.add(name=f'exclude{index}', region=region.name, 

138 kind=schema.CapacitanceExclusion.KIND_ALL_LAYER_PAIRS) 

139 else: 

140 first, second = sorted(block.capacitance_pair) 

141 for ordinal, (a, b) in enumerate(itertools.product( 

142 [name for name, alias in aliases.items() if alias == first], 

143 [name for name, alias in aliases.items() if alias == second])): 

144 policy.exclusions.add(name=f'exclude{index}_{ordinal}', region=region.name, 

145 kind=schema.CapacitanceExclusion.KIND_LAYER_PAIR, 

146 layer_a=a, layer_b=b) 

147 

148 if not (policy.regions or policy.exclusions or policy.excluded_substrate_nets): 

149 # An unused marker or excluded name must not manufacture a required 

150 # accounting contract for an otherwise ordinary geometry scene. 

151 file.ClearField('capacitance_accounting') 

152 file.ClearField('required_capabilities') 

153 from ..pex25d.format_version import FORMAT_VERSION_SUFFIX 

154 file.format_version_suffix = FORMAT_VERSION_SUFFIX
