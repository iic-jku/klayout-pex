53def read_preparation(path: Path) -> PreparedInput: 

54 from .patterns import read_patterns 

55 from ..tech_identity import message_sha256 

56 from klayout_pex_protobuf.kpex.calibration.preparation_pb2 import CalibrationManifest, CalibrationCase 

57 from klayout_pex_protobuf.kpex.calibration.preparation_pb2 import CapacitanceMeasurement as Measurement 

58 from klayout_pex_protobuf.kpex.calibration.patterns_pb2 import SUBSTRATE 

59 from klayout_pex_protobuf.kpex.pex25d.pex25d_units_pb2 import Units 

60 from klayout_pex_protobuf.kpex.tech.tech_pb2 import Technology 

61 

62 digest = sha256_file(path) 

63 manifest = read_message(path, CalibrationManifest()) 

64 if (manifest.schema_version != 1 or manifest.pattern_generator_version != 1 

65 or manifest.stage != CalibrationManifest.PREPARED or not manifest.kpex_version): 

66 raise CalibrationError('Unsupported preparation version or stage') 

67 verify_artifacts(path.parent, manifest.artifacts) 

68 spec = read_patterns(artifact_path(path.parent, 'patterns.json')) 

69 tech = read_message(artifact_path(path.parent, 'technology.pb.json'), Technology()) 

70 if (manifest.source_specification_sha256 != spec.source_sha256 

71 or manifest.technology.name != tech.name 

72 or manifest.technology.resolved_technology_sha256 != message_sha256(tech) 

73 or manifest.technology.physical_stack_sha256 != message_sha256(tech.process_stack)): 

74 raise CalibrationError('Preparation source or technology identity differs from its snapshot') 

75 if manifest.technology.HasField('selection'): 

76 selection = manifest.technology.selection 

77 if (selection.technology_id != tech.name or not selection.scenario_id 

78 or selection.resolved_technology_sha256 != manifest.technology.resolved_technology_sha256 

79 or not selection.physical_stack_matches 

80 or re.fullmatch(r'[0-9a-f]{64}', selection.catalog_sha256) is None): 

81 raise CalibrationError('Preparation catalog selection differs from its resolved technology') 

82 if (manifest.units != Units(length=Units.LENGTH_UNIT_UM, grid_numerator=spec.grid.numerator, 

83 grid_denominator=spec.grid.denominator) 

84 or manifest.domain_margin_grid != spec.domain_margin 

85 or list(manifest.targets) != [target.definition for target in spec.targets] 

86 or manifest.fringe_scale_mode != spec.fringe_scale_mode 

87 or manifest.technology_side_halo_um != tech.process_parasitics.side_halo): 

88 raise CalibrationError('Preparation units, targets, halo or fringe scale differ from its source') 

89 if len(manifest.cases) != len(spec.patterns): 

90 raise CalibrationError('Preparation case count differs from its source') 

91 expected_paths = {'patterns.json', 'technology.pb.json'} 

92 for case, pattern in zip(manifest.cases, spec.patterns): 

93 expected = CalibrationCase(id=pattern.case_id, family=pattern.family, role=pattern.role, 

94 layers=pattern.layers, target_ids=pattern.target_ids, 

95 file=f'case_{pattern.case_id}.pex25d', 

96 scene=f'case_{pattern.case_id}.pex25d.scene.pb') 

97 expected.dimensions_grid.CopyFrom(type(case.dimensions_grid)( 

98 width=pattern.width, length=pattern.length, gap=pattern.gap, 

99 receiver_width=pattern.receiver_width, lateral_offset=pattern.lateral_offset)) 

100 expected.measurement.CopyFrom(Measurement( 

101 kind=Measurement.MAXWELL_CAPACITANCE_MATRIX, 

102 conductor_order=['A'] if pattern.family == SUBSTRATE else ['A', 'B'], 

103 capacitance_unit=Measurement.FARAD, convention=Measurement.MAXWELL_GROUNDED_UNEXCITED)) 

104 reference = case.measurement.reference.profile 

105 if not reference or reference in expected.measurement.conductor_order: 

106 raise CalibrationError('Preparation needs a distinct finite reference conductor') 

107 expected.measurement.reference.profile = reference 

108 expected.diagnostics.extend(case.diagnostics) 

109 if case != expected: 

110 raise CalibrationError(f"Prepared case '{case.id}' differs from its specification") 

111 expected_paths.update((case.file, case.scene)) 

112 if {record.path for record in manifest.artifacts} != expected_paths: 

113 raise CalibrationError('Preparation artifact set differs from its specification') 

114 if sha256_file(path) != digest: 

115 raise CalibrationError('Preparation changed while being read') 

116 return PreparedInput(path.resolve(), digest, manifest, spec, tech)

119def validated_scenes(prepared: PreparedInput) -> List[PEX25DScene]: 

120 from .prepare import _case_file, _resolve_case, _stack 

121 from .. import pex25d 

122 from ..tech_info import TechInfo 

123 

124 try: 

125 from .targets import validate_target_rows 

126 tech = TechInfo(prepared.technology, dielectric_filter=None) 

127 validate_target_rows(tech, prepared.specification.targets) 

128 stack = _stack(tech, prepared.specification) 

129 scenes = [] 

130 for case, pattern in zip(prepared.manifest.cases, prepared.specification.patterns): 

131 file = pex25d.read(str(artifact_path(prepared.path.parent, case.file))) 

132 scene = pex25d.read(str(artifact_path(prepared.path.parent, case.scene))) 

133 expected, _ = _resolve_case(_case_file(stack, pattern), pattern) 

134 resolved, _ = _resolve_case(file, pattern) 

135 for message in (scene, expected, resolved): 

136 message.ClearField('metadata') 

137 if scene != expected or scene != resolved or scene.ground_plane.name != case.measurement.reference.profile: 

138 raise CalibrationError(f"Prepared case '{case.id}' geometry differs from its specification") 

139 scenes.append(scene) 

140 return scenes 

141 except CalibrationError: 

142 raise 

143 except Exception as exc: 

144 raise CalibrationError(f'Cannot validate prepared geometry: {exc}') from exc
