51@dataclass(frozen=True) 

52class QualificationRun: 

53 root: Path 

54 run_sha256: str 

55 collection: CalibrationCollection 

56 manifest: CalibrationRunManifest 

57 preparation: 'CalibrationManifest' 

58 artifacts: Dict[str, str] 

59 

60 def artifact(self, path: Path) -> Path: 

61 name = path.relative_to(self.root).as_posix() 

62 if name not in self.artifacts: 

63 raise CalibrationError(f"'{path}' is outside the saved collection artifact set") 

64 verified = artifact_path(self.root, name) 

65 if sha256_file(verified) != self.artifacts[name]: 

66 raise CalibrationError(f"Collection artifact '{name}' has changed") 

67 return verified

70@dataclass(frozen=True) 

71class QualificationMeasurement: 

72 run: QualificationRun 

73 pattern: 'CalibrationCase' 

74 record: CalibrationCaseResult 

75 scene: PEX25DScene 

76 solver_inputs: Tuple[Tuple[str, str], ...]

78 

79def read_collection(path: Path, expected_hash: str) -> QualificationRun: 

80 root = path.parent 

81 collection = read_message(path, CalibrationCollection()) 

82 if sha256_file(path) != expected_hash or collection.schema_version != 1: 

83 raise CalibrationError(f"Collection '{path}' changed or has an unsupported schema") 

84 verify_artifacts(root, collection.artifacts) 

85 hashes = {item.path: item.sha256 for item in collection.artifacts} 

86 run_path = artifact_path(root, collection.run_manifest.path) 

87 if hashes.get(collection.run_manifest.path) != collection.run_manifest.sha256: 

88 raise CalibrationError('Run manifest is absent from the collection artifact set') 

89 run = read_message(run_path, CalibrationRunManifest()) 

90 if (run.schema_version != 1 or not run.kpex_version or run.solver.name != 'fastercap' 

91 or not run.solver.version or not run.solver.executable_sha256 

92 or not run.solver.operating_system or not run.solver.architecture): 

93 raise CalibrationError('Unsupported calibration run schema or solver identity') 

94 try: 

95 version = Version(run.solver.version) 

96 except InvalidVersion as exc: 

97 raise CalibrationError(f"Invalid recorded FasterCap version '{run.solver.version}'") from exc 

98 supported = effective_version_range(Tool.FASTERCAP) 

99 if version not in supported: 

100 raise CalibrationError(f'Qualification requires FasterCap {supported}; the saved run used {version}') 

101 validate_options_message(run.options) 

102 

103 prepared_path = artifact_path(run_path.parent, run.preparation.path) 

104 if hashes.get(prepared_path.relative_to(root).as_posix()) != run.preparation.sha256: 

105 raise CalibrationError('Preparation is absent from the collection artifact set') 

106 prepared = read_preparation(prepared_path) 

107 if prepared.sha256 != run.preparation.sha256: 

108 raise CalibrationError('Preparation changed during qualification') 

109 for artifact in prepared.manifest.artifacts: 

110 name = (prepared_path.parent.relative_to(root) / artifact.path).as_posix() 

111 if hashes.get(name) != artifact.sha256: 

112 raise CalibrationError(f"Prepared artifact '{name}' is absent from the collection") 

113 case_ids = [case.case_id for case in collection.cases] 

114 if (not case_ids or len({case_id.casefold() for case_id in case_ids}) != len(case_ids) 

115 or case_ids != [case.case_id for case in run.cases] 

116 or case_ids != [case.id for case in prepared.manifest.cases]): 

117 raise CalibrationError('Collection case IDs differ from the run or preparation') 

118 return QualificationRun(root, collection.run_manifest.sha256, 

119 collection, run, prepared.manifest, hashes)

122def read_measurement(run: QualificationRun, case_id: str) -> QualificationMeasurement: 

123 from ..fastercap.result_collection import collect_fastercap_matrix 

124 

125 index = next((i for i, case in enumerate(run.collection.cases) if case.case_id == case_id), None) 

126 if index is None: 

127 raise CalibrationError(f"Referenced collection does not contain case '{case_id}'") 

128 pattern = run.preparation.cases[index] 

129 record = run.collection.cases[index] 

130 case = run.manifest.cases[index] 

131 run_root = (run.root / run.collection.run_manifest.path).parent 

132 prepared_root = artifact_path(run_root, run.manifest.preparation.path).parent 

133 scene_path = run.artifact(artifact_path(prepared_root, pattern.scene)) 

134 try: 

135 scene = PEX25DScene.FromString(scene_path.read_bytes()) 

136 except (DecodeError, OSError) as exc: 

137 raise CalibrationError(f"Invalid saved scene for '{case_id}': {exc}") from exc 

138 scene.ClearField('metadata') 

139 domain = scene.domain 

140 margin = run.preparation.domain_margin_grid 

141 box, bounds = domain.box, domain.geometry_bounds 

142 if (scene.units != run.preparation.units or scene.ground_plane.name != pattern.measurement.reference.profile 

143 or domain.origin != ResolvedDomain.ORIGIN_DOMAIN_MARGIN or not domain.HasField('geometry_bounds') 

144 or not domain.HasField('box') or not domain.HasField('applied_margin') 

145 or (domain.applied_margin.x, domain.applied_margin.y, domain.applied_margin.z) != (margin,) * 3 

146 or (box.lower_left.x, box.lower_left.y, box.lower_left.z, 

147 box.upper_right.x, box.upper_right.y, box.upper_right.z) != ( 

148 bounds.lower_left.x - margin, bounds.lower_left.y - margin, scene.ground_plane.zhigh, 

149 bounds.upper_right.x + margin, bounds.upper_right.y + margin, bounds.upper_right.z + margin)): 

150 raise CalibrationError(f"Saved geometry or domain for '{case_id}' differs from the preparation") 

151 

152 if record.status == CalibrationCaseResult.COLLECTED: 

153 checkpoint = read_message(run.artifact(artifact_path(run_root, case.checkpoint_path)), CalibrationRunCheckpoint()) 

154 if (checkpoint.schema_version != 1 or checkpoint.case_id != case_id 

155 or checkpoint.execution != record.execution or record.execution.status != SolverExecution.SUCCEEDED 

156 or not record.execution.HasField('return_code') or record.execution.return_code != 0 

157 or not record.execution.arguments): 

158 raise CalibrationError(f"Collected case '{case_id}' does not identify a successful solver execution") 

159 if (not case.numerical_report_path or checkpoint.numerical_report.path != case.numerical_report_path 

160 or checkpoint.log.path != case.log_path): 

161 raise CalibrationError(f"Collected case '{case_id}' requires the full-precision FasterCap report") 

162 log = run.artifact(artifact_path(run_root, case.log_path)) 

163 report = run.artifact(artifact_path(run_root, case.numerical_report_path)) 

164 log_name = log.relative_to(run.root).as_posix() 

165 report_name = report.relative_to(run.root).as_posix() 

166 if (run.artifacts.get(log_name) != checkpoint.log.sha256 

167 or run.artifacts.get(report_name) != checkpoint.numerical_report.sha256): 

168 raise CalibrationError(f"Collected case '{case_id}' has inconsistent checkpoint hashes") 

169 actual_input = record.execution.arguments[-1].removeprefix('"').removesuffix('"') 

170 try: 

171 native, measured = collect_fastercap_matrix( 

172 log, list(pattern.measurement.conductor_order), pattern.measurement.reference.profile, 

173 numerical_report=report, 

174 expected_input=actual_input, expected_version=run.manifest.solver.version, 

175 ) 

176 except FasterCapError as exc: 

177 raise CalibrationError(f"Cannot verify collected matrix for '{case_id}': {exc}") from exc 

178 panels = read_native_statistics(log)['refined_panels'] 

179 if panels is not None and panels > 2**64 - 1: 

180 panels = None 

181 if (record.refined_panels if record.HasField('refined_panels') else None) != panels: 

182 raise CalibrationError(f"Collected panel count for '{case_id}' differs from the saved log") 

183 if record.native_matrix != native or record.measured_matrix != measured: 

184 raise CalibrationError(f"Collected matrix for '{case_id}' differs from the native report") 

185 elif record.status not in (CalibrationCaseResult.INCOMPLETE, CalibrationCaseResult.EXECUTION_FAILED, 

186 CalibrationCaseResult.INVALID_RESULT): 

187 raise CalibrationError(f"Invalid collection status for '{case_id}'") 

188 inputs = [] 

189 for item in case.inputs: 

190 path = run.artifact(artifact_path(run_root, item.path)) 

191 if run.artifacts[path.relative_to(run.root).as_posix()] != item.sha256: 

192 raise CalibrationError(f"Solver input for '{case_id}' differs from the run manifest") 

193 inputs.append((item.path, item.sha256)) 

194 return QualificationMeasurement(run, pattern, record, scene, tuple(sorted(inputs)))
