38class FasterCapInputBuilder: 

39 """Layout-side preparation; geometry export is shared with standalone scenes.""" 

40 

41 def __init__(self, pex_context: KLayoutExtractionContext, tech_info: TechInfo, 

42 substrate_net_name: str, k_void: float = 3.5, 

43 delaunay_amax: float = 0.0, delaunay_b: float = 1.0, 

44 exclusions: Optional[Exclusions] = None): 

45 self.pex_context = pex_context 

46 self.tech_info = tech_info 

47 self.substrate_net_name = substrate_net_name 

48 self.k_void = k_void 

49 self.delaunay_amax = delaunay_amax 

50 self.delaunay_b = delaunay_b 

51 self.exclusions = exclusions 

52 

53 def build_scene(self) -> Any: 

54 from ..klayout.pex25d_builder import PEX25DBuilder, BuilderOptions, GeometryScope, BuildError 

55 from ..klayout.pex25d_exclusions import attach_layout_accounting 

56 from ..pex25d import resolve, ResolveError 

57 from ..klayout.physical_geometry import PhysicalGeometryError 

58 from ..klayout.pex25d_ground import GroundGeometryError 

59 from ..pex25d.accounting import SceneCapacitanceMap 

60 from fractions import Fraction 

61 from ..klayout.pex25d_accounting import AccountingGridError 

62 from ..klayout.pex25d_builder import DEFAULT_GRID_UM 

63 grid = Fraction(DEFAULT_GRID_UM) 

64 try: 

65 while True: 

66 builder = PEX25DBuilder( 

67 self.pex_context, self.tech_info, self.pex_context.top_circuit.name, 

68 BuilderOptions(grid_um=str(grid), geometry_scope=GeometryScope.ELECTROSTATIC, 

69 include_terminals=False, include_resistance=False, 

70 background_permittivity=self.k_void, domain_margin_um=8.0)) 

71 file = builder.build() 

72 file.ground_plane.net = self.substrate_net_name 

73 if self.exclusions is not None: 

74 attach_layout_accounting(file, builder, self.exclusions) 

75 try: 

76 scene = resolve(file) 

77 break 

78 except ResolveError as exc: 

79 cause = exc.__cause__ 

80 if not isinstance(cause, AccountingGridError): 

81 raise 

82 # Rebuild from the original source grid; repeated rescaling 

83 # of rounded geometry would change via ownership and corners. 

84 grid /= cause.refinement_factor 

85 except (ValueError, BuildError, ResolveError, PhysicalGeometryError, GroundGeometryError) as exc: 

86 raise FasterCapError(f'Cannot build physical FasterCap scene: {exc}') from exc 

87 self.scene = scene 

88 self.accounting = SceneCapacitanceMap(scene) 

89 if self.exclusions is not None: 

90 # Preserve the existing enclosing netlist API for callers that still 

91 # pass Exclusions. Geometry/export itself uses only the scene map. 

92 for name, part in self.accounting.parts.items(): 

93 indexes = frozenset(int(r.removeprefix('block')) for r in part.regions) 

94 layer = part.layer 

95 if layer is not None: 

96 pair = self.tech_info.gds_pair(layer) 

97 layer = self.tech_info.canonical_layer_name_by_gds_pair.get(pair, layer) 

98 original = ConductorPart(part.body.name, indexes, layer, part.is_reference) 

99 self.exclusions.conductor_parts[name] = original 

100 self.exclusions.conductor_names[original] = name 

101 return scene

103 def build(self) -> FasterCapModelGenerator: 

104 from .pex25d_exporter import build_fastercap_model, FasterCapModelOptions 

105 from ..plugin_api.v1 import ExportError 

106 try: 

107 return build_fastercap_model(self.build_scene(), 

108 FasterCapModelOptions(delaunay_amax=self.delaunay_amax, 

109 delaunay_b=self.delaunay_b), 'FasterCap') 

110 except ExportError as exc: 

111 raise FasterCapError(f'Cannot export physical FasterCap scene: {exc}') from exc
