41def add_execution_subparsers(commands: argparse._SubParsersAction): 

42 solve = commands.add_parser( 

43 'solve', help='Run prepared cases and collect raw FasterCap matrices', 

44 description='Export cases, run FasterCap and collect raw matrices automatically. ' 

45 'A successful run is not a numerical qualification.', 

46 formatter_class=RichHelpFormatter, allow_abbrev=False) 

47 solve.add_argument('--manifest', required=True, type=Path, metavar='FILE', 

48 help='Input CalibrationManifest from calibrate prepare (protobuf JSON)') 

49 execution = solve.add_argument_group('Execution controls') 

50 execution.add_argument('--solver', choices=['fastercap'], default='fastercap', 

51 help='Solver backend (default: fastercap)') 

52 execution.add_argument('--fastercap-exe', default=None, metavar='PATH', 

53 help='Executable name or path (default: KPEX_FASTERCAP_EXE, then FasterCap)') 

54 execution.add_argument('--jobs', type=int, default=1, metavar='N', 

55 help='Concurrent solver processes, 1–32 (default: 1); no memory limit is imposed') 

56 execution.add_argument('--threads-per-job', type=int, default=1, metavar='N', 

57 help='OpenMP/numerical-library threads requested per solver process, 1–256 ' 

58 '(default: 1); support depends on the solver build') 

59 execution.add_argument('--solver-timeout-seconds', type=positive_number, default=600, metavar='SECONDS', 

60 help='Wall-clock deadline per solver process (default: 600); ' 

61 'excludes queueing and geometry export; timed-out evidence is retained') 

62 numerical = solve.add_argument_group('FasterCap numerical controls') 

63 numerical.add_argument('--fastercap-auto-tolerance', type=relative_error, metavar='FRACTION', 

64 help='Automatic refinement stopping tolerance, 0 < value < 1 (default: 0.05). ' 

65 'Cannot be combined with manual mesh/GMRES controls; ' 

66 'not a guaranteed capacitance error bound') 

67 numerical.add_argument('--fastercap-mesh-refinement', type=positive_number, metavar='VALUE', 

68 help='Manual relative refinement parameter (-m), > 0; requires ' 

69 '--fastercap-gmres-tolerance and excludes --fastercap-auto-tolerance') 

70 numerical.add_argument('--fastercap-gmres-tolerance', type=relative_error, metavar='FRACTION', 

71 help='Manual iterative-solver tolerance (-t), 0 < value < 1; requires ' 

72 '--fastercap-mesh-refinement and excludes --fastercap-auto-tolerance') 

73 numerical.add_argument('--fastercap-interaction-ratio', type=positive_number, default=0.5, metavar='RATIO', 

74 help='Direct potential interaction coefficient / mesh refinement (-d), ' 

75 '> 0 (default: 0.5); applies in both refinement modes') 

76 mesh = solve.add_argument_group('Initial geometry mesh') 

77 mesh.add_argument('--mesh-max-area-um2', type=non_negative_square_micrometers, default=0, metavar='AREA', 

78 help='Maximum initial triangle area in µm², >= 0 (default: 0, mesher chooses); ' 

79 'independent of FasterCap adaptive refinement') 

80 mesh.add_argument('--mesh-min-angle-deg', type=float, default=30, metavar='DEGREES', 

81 help='Requested minimum initial triangle angle in degrees, 0–about 33.367 ' 

82 '(default: 30); preserves the mesher quality limit') 

83 collect = commands.add_parser( 

84 'collect', help='Collect saved solver logs without running a solver', 

85 description='Read recorded checkpoints and logs, validate their hashes, and collect raw matrices. ' 

86 'No solver installation is needed. Failed or incomplete cases remain explicit.', 

87 formatter_class=RichHelpFormatter, allow_abbrev=False) 

88 collect.add_argument('--manifest', required=True, type=Path, metavar='FILE', 

89 help='Input CalibrationRunManifest from calibrate solve (protobuf JSON)') 

90 for parser in (solve, collect): 

91 parser.add_argument('--out-dir', required=True, type=Path, metavar='DIR', 

92 help='New output directory under an existing parent; existing paths are refused') 

93 add_log_level_argument(parser)

96def run_execution_command(args: argparse.Namespace): 

97 from .collect import collect_calibration 

98 from .solve import SolveOptions, solve_calibration 

99 import os 

100 

101 try: 

102 if args.calibration_command == 'solve': 

103 options = SolveOptions(jobs=args.jobs, threads_per_job=args.threads_per_job, 

104 solver_timeout_seconds=args.solver_timeout_seconds, 

105 automatic_tolerance=args.fastercap_auto_tolerance, 

106 mesh_refinement=args.fastercap_mesh_refinement, 

107 gmres_tolerance=args.fastercap_gmres_tolerance, 

108 interaction_ratio=args.fastercap_interaction_ratio, 

109 mesh_max_area_um2=args.mesh_max_area_um2, mesh_min_angle_deg=args.mesh_min_angle_deg) 

110 executable = args.fastercap_exe or os.environ.get(EnvVar.FASTERCAP_EXE, 

111 EnvVar.FASTERCAP_EXE.default_value) 

112 result = solve_calibration(args.manifest, args.out_dir, executable=executable, options=options) 

113 info(f'Run manifest: {result.manifest_path}') 

114 path = result.collection_path 

115 elif args.calibration_command == 'collect': 

116 result = collect_calibration(args.manifest, args.out_dir) 

117 path = result.manifest_path 

118 else: 

119 raise NotImplementedError(f'Calibration execution command {args.calibration_command!r}') 

120 except (CalibrationError, OSError) as exc: 

121 error(str(exc)) 

122 sys.exit(ExitCode.USAGE) 

123 info(f'Raw collection: {result.successful_cases} collected, {result.failed_cases} failed or incomplete: {path}') 

124 info('Numerical qualification and coefficient fitting have not run') 

125 if result.failed_cases: 

126 sys.exit(ExitCode.DIAGNOSTIC_ERRORS)
