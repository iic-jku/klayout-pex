42EPILOG = """ 

43Select extracted elements explicitly. Defaults select `Cext_*`, `Cgrnd_*` and `Rext_*`. 

44`--capacitor` and `--resistor` replace their respective defaults and may be repeated. 

45Patterns and SPICE node names are case-insensitive. Quote patterns to prevent shell expansion. 

46Selected elements must have exactly two nodes and a finite numeric value. Unselected devices, 

47resistors, comments, directives and other scopes remain verbatim. 

48 

49Selected resistors define distributed conductors without being shorted or rewritten. Subcircuit 

50ports, global nodes, terminals on unselected device/source lines and `--keep` nodes retain their 

51conductors. All other conductors are floating. A flat `--net-map` JSON object such as 

52`{"A1": "A", "A2": "A"}` can identify disconnected pieces of the same conductor. 

53It cannot assign different conductors to nodes connected by selected resistors. 

54 

55Capacitor-only floating nodes are eliminated exactly before grounding. Floating resistive islands 

56and elimination exceeding the shared 64-neighbor limit fail without writing output. 

57Each remaining inter-net capacitor becomes its full value to `--reference` at both ends. 

58Capacitance within a conductor, including distributed reference nodes, stays at its original ends. 

59The reference is explicit and must already be a node or port in the selected scope, or `0`/`gnd`. 

60 

61This intentionally removes inter-net transfer/crosstalk. It preserves diagonal capacitance after 

62floating elimination, not arbitrary distributed-RC port impedance or switching-dependent Miller effects. 

63Use a coupled netlist with intentional capacitors represented as retained devices. Capacitance from 

64a white-boxed device cannot be distinguished from parasitics after extraction. 

65 

66| Exit code | Meaning | 

67| --------- | ------- | 

68| 0 | grounded netlist written | 

69| 1 | invalid selected elements, malformed netlist, or unsupported floating network | 

70| 2 | invalid options, ambiguous scope, unknown nodes, or file I/O failure | 

71"""

74def add_ground_subcommand(subparsers: argparse._SubParsersAction) -> argparse.ArgumentParser: 

75 parser = subparsers.add_parser( 

76 'ground', help='Ground selected parasitic coupling after exact floating-capacitance elimination', 

77 description='Ground inter-net capacitance in a SPICE netlist without layout or technology data.', 

78 formatter_class=RichHelpFormatter, 

79 epilog=rich.console.Group(rich.text.Text('Selection, approximation and exit codes:', style='argparse.groups'), 

80 rich.markdown.Markdown(EPILOG, style='argparse.text'))) 

81 parser.add_argument('input_netlist_path', metavar='INPUT_NETLIST') 

82 parser.add_argument('--out-netlist', '-o', dest='output_netlist_path', metavar='OUTPUT_NETLIST', 

83 help="Output netlist (default: INPUT_NETLIST with '_grounded' before its extension)") 

84 parser.add_argument('--reference', required=True, metavar='NODE', help='Reference for new grounded loads') 

85 scope = parser.add_mutually_exclusive_group() 

86 scope.add_argument('--cell', dest='cell_name', metavar='NAME', help='Subcircuit to transform') 

87 scope.add_argument('--top-level', action='store_true', help='Transform only elements outside subcircuits') 

88 parser.add_argument('--capacitor', action='append', metavar='PATTERN', 

89 help='Parasitic capacitor name pattern (default: Cext_* and Cgrnd_*)') 

90 parser.add_argument('--resistor', action='append', metavar='PATTERN', 

91 help='Parasitic resistor name pattern (default: Rext_*)') 

92 parser.add_argument('--keep', action='append', default=[], metavar='NODE', 

93 help='Retain this node and its conductor (repeatable)') 

94 parser.add_argument('--net-map', metavar='JSON', help='Flat node-to-conductor map extending inferred identities') 

95 return parser

98def run_ground_subcommand(args: argparse.Namespace) -> ExitCode: 

99 try: 

100 options = GroundingOptions( 

101 reference=args.reference, 

102 capacitor_patterns=tuple(args.capacitor) if args.capacitor is not None else ('Cext_*', 'Cgrnd_*'), 

103 resistor_patterns=tuple(args.resistor) if args.resistor is not None else ('Rext_*',), 

104 keep_nodes=tuple(args.keep), cell_name=args.cell_name, top_level=args.top_level) 

105 if args.net_map is not None: 

106 options = replace(options, net_by_node=read_net_map(args.net_map)) 

107 result = ground_spice_netlist(args.input_netlist_path, 

108 args.output_netlist_path or grounded_netlist_path(args.input_netlist_path), 

109 options) 

110 except (CouplingPolicyError, NetlistReadError) as exc: 

111 error(str(exc)) 

112 error('Keep the coupled input for unsupported networks, or adjust explicit parasitic selection') 

113 return ExitCode.DIAGNOSTIC_ERRORS 

114 except (ValueError, OSError) as exc: 

115 error(str(exc)) 

116 return ExitCode.USAGE 

117 info(f'{result.scope}: {result.capacitors_read} selected capacitors -> {result.capacitors_written} grounded capacitors') 

118 if result.floating_nodes: 

119 warning('Eliminated floating nodes: ' + ', '.join(result.floating_nodes) 

120 + '. Verify these conductors have no external connection') 

121 warning('Grounding removes inter-net transfer admittance and preserves reduced nodal capacitance, ' 

122 'not arbitrary distributed-RC port impedance') 

123 info(f'Wrote grounded netlist to {result.output_path}') 

124 return ExitCode.OK
