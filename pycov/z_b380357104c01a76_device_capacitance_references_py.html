45class DeviceCapacitanceCheckError(Exception): 

46 """A requested device-capacitance comparison cannot be evaluated."""

49def _number(value: Any, what: str, allow_suffix: bool = False) -> float: 

50 if allow_suffix and isinstance(value, str): 

51 value = parse_spice_number(value) 

52 if isinstance(value, bool) or not isinstance(value, (int, float)) or not math.isfinite(value): 

53 raise ValueError(f'{what} must be a finite number') 

54 return float(value)

57@dataclass(frozen=True) 

58class DeviceCapacitanceReference: 

59 device: str 

60 device_class: str 

61 terminals: Tuple[str, str] 

62 capacitance_f: float 

63 min_ratio: Optional[float] = None 

64 max_ratio: Optional[float] = None 

65 source: Optional[str] = None 

66 description: Optional[str] = None 

67 

68 def __post_init__(self): 

69 if _number(self.capacitance_f, 'capacitance') <= 0: 

70 raise ValueError('capacitance must be greater than zero') 

71 for name, bound in (('min_ratio', self.min_ratio), ('max_ratio', self.max_ratio)): 

72 if bound is not None and _number(bound, name) < 0: 

73 raise ValueError(f'{name} must be nonnegative') 

74 if self.min_ratio is not None and self.max_ratio is not None and self.min_ratio > self.max_ratio: 

75 raise ValueError('ratio bounds must have min_ratio <= max_ratio')

78def read_device_capacitance_references(path: str) -> List[DeviceCapacitanceReference]: 

79 """Read version 1 references; numeric capacitances are in F, strings accept SPICE suffixes.""" 

80 def unique_object(pairs: List[Tuple[str, Any]]) -> Dict[str, Any]: 

81 result = {} 

82 for key, value in pairs: 

83 if key in result: 

84 raise ValueError(f'duplicate JSON key {key!r}') 

85 result[key] = value 

86 return result 

87 

88 def fields(value: Any, required: Set[str], optional: Set[str], where: str): 

89 if not isinstance(value, dict) or not required <= value.keys() or value.keys() - required - optional: 

90 raise ValueError(f'{where} needs {", ".join(sorted(required))}; ' 

91 f'optional fields: {", ".join(sorted(optional)) or "none"}') 

92 

93 try: 

94 with open(path, encoding='utf-8') as stream: 

95 document = json.load(stream, object_pairs_hook=unique_object) 

96 fields(document, {'format_version', 'references'}, set(), 'reference file') 

97 if type(document['format_version']) is not int or document['format_version'] != 1: 

98 raise ValueError('format_version must be 1') 

99 if not isinstance(document['references'], list) or not document['references']: 

100 raise ValueError('references must be a nonempty list') 

101 references = [] 

102 seen = set() 

103 for index, item in enumerate(document['references']): 

104 where = f'reference {index + 1}' 

105 fields(item, {'device', 'device_class', 'terminals', 'capacitance'}, 

106 {'min_ratio', 'max_ratio'}, where) 

107 for name in ('device', 'device_class'): 

108 if not isinstance(item[name], str) or not item[name].strip(): 

109 raise ValueError(f'{where}: {name} must be a nonempty string') 

110 terminals = item['terminals'] 

111 if not isinstance(terminals, list) or len(terminals) != 2 or \ 

112 any(not isinstance(name, str) or not name.strip() for name in terminals) or \ 

113 terminals[0] == terminals[1]: 

114 raise ValueError(f'{where}: terminals must name two different terminals') 

115 capacitance = _number(item['capacitance'], f'{where}: capacitance', allow_suffix=True) 

116 low = _number(item['min_ratio'], f'{where}: min_ratio') if 'min_ratio' in item else None 

117 high = _number(item['max_ratio'], f'{where}: max_ratio') if 'max_ratio' in item else None 

118 key = item['device'], frozenset(terminals) 

119 if key in seen: 

120 raise ValueError(f'{where}: duplicate device and terminal pair') 

121 seen.add(key) 

122 references.append(DeviceCapacitanceReference(item['device'], item['device_class'], 

123 tuple(terminals), capacitance, low, high)) 

124 return references 

125 except (OSError, UnicodeError, ValueError, OverflowError) as exc: 

126 raise DeviceCapacitanceCheckError(f'Invalid device-capacitance reference file {path}: {exc}') from exc

129def tech_device_capacitance_references(context: KLayoutExtractionContext) -> Tuple[ 

130 List[DeviceCapacitanceReference], List[str]]: 

131 """Evaluate tech references and name metal capacitors without one; never use LVS C placeholders.""" 

132 models = context.tech.device_models 

133 references: List[DeviceCapacitanceReference] = [] 

134 unchecked: List[str] = [] 

135 parser = ExpressionParser(NGSPICE_SYNTAX) 

136 for device in context.top_circuit.each_device(): 

137 name = device.expanded_name() 

138 dc = device.device_class() 

139 if dc.name not in models.metal_capacitor_class_names: 

140 continue 

141 mapping = models.mapping_by_lvs_device_class_name[dc.name] 

142 if not mapping.capacitance_references: 

143 unchecked.append(f'{name} ({dc.name})') 

144 continue 

145 try: 

146 names = [p.name.lower() for p in mapping.parameters] 

147 if len(names) != len(set(names)): 

148 raise ValueError('duplicate model parameter names') 

149 values = {key.lower(): _number(value, key) for key, value in 

150 models.model_parameter_values(mapping, device).items()} 

151 terminals = {terminal.name for terminal in dc.terminal_definitions()} 

152 pairs = set() 

153 for spec in mapping.capacitance_references: 

154 pair = frozenset(spec.terminal_names) 

155 if len(spec.terminal_names) != 2 or len(pair) != 2 or not pair <= terminals: 

156 raise ValueError('reference needs two distinct LVS terminals; valid terminals: ' 

157 + ', '.join(sorted(terminals))) 

158 if pair in pairs: 

159 raise ValueError('duplicate reference terminal pair') 

160 pairs.add(pair) 

161 if not spec.source.strip() or not spec.description.strip(): 

162 raise ValueError('reference needs source and nominal assumptions in description') 

163 local_names = [key.lower() for key in spec.parameters] 

164 if len(local_names) != len(set(local_names)) or set(local_names) & values.keys(): 

165 raise ValueError('local parameter names must be unique and must not shadow model parameters') 

166 scope = ParameterScope({key.lower(): parser.parse(value) for key, value in spec.parameters.items()}, 

167 parent=ParameterScope(values)) 

168 # Validate all definitions, including ones an inactive conditional would not evaluate. 

169 for key in local_names: 

170 scope.value(key) 

171 if spec.valid_when and not scope.evaluate(parser.parse(spec.valid_when)): 

172 raise ValueError(f'outside reference domain: {spec.valid_when}') 

173 capacitance = scope.evaluate(parser.parse(spec.capacitance)) 

174 references.append(DeviceCapacitanceReference( 

175 name, dc.name, tuple(spec.terminal_names), capacitance, 

176 source=spec.source, description=spec.description)) 

177 except (ExpressionError, DeviceModelError, ValueError, RuntimeError) as exc: 

178 raise DeviceCapacitanceCheckError( 

179 f'Tech-info capacitance reference for {name} ({dc.name}): {exc}') from exc 

180 if not references: 

181 raise DeviceCapacitanceCheckError('No tech-info capacitance references for this layout. ' 

182 'Supply --check-device-caps JSON with explicit device references. ' 

183 'Unchecked metal capacitors: ' + (', '.join(unchecked) or 'none')) 

184 return references, unchecked
