96def test_nfet_10v0_asym_pair(): 

97 # 2 mirrored LDMOS sharing their drain (the MVSD side), each with its own source: 

98 # the asymmetric device must be extracted with S and D in place, not swapped 

99 cell_name = 'nfet_10v0_asym_pair' 

100 _, csv_path, _ = pex_whiteboxed.run_rcx25d_single_cell('test_patterns', f"{cell_name}.gds.gz") 

101 lvsdb = kdb.LayoutVsSchematic() 

102 lvsdb.read(os.path.join(os.path.dirname(csv_path), f"{cell_name}.lvsdb.gz")) 

103 devices = list(lvsdb.netlist().circuit_by_name(cell_name).each_device()) 

104 assert [d.device_class().name for d in devices] == ['nfet_10v0_asym'] * 2 

105 drains = {d.net_for_terminal('D').expanded_name() for d in devices} 

106 sources = {d.net_for_terminal('S').expanded_name() for d in devices} 

107 assert len(drains) == 1 

108 assert len(sources) == 2
