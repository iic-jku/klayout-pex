110class ConnectedShapesWithin: 

111 """ 

112 Which shapes of a net within the blocks are joined there, by metal or vias, regardless of the rest of the net, 

113 so that a label on them can name the ports of a cut of only these shapes 

114 """ 

115 def __init__(self, 

116 regions_within: Mapping[int, Mapping[LayerName, kdb.Region]], 

117 vias_within: Iterable[Tuple[LayerName, LayerName, kdb.Region]], 

118 kept_for_vias: Mapping[LayerName, kdb.Region]): 

119 self.kept_for_vias = kept_for_vias 

120 self.shapes_by_layer: Dict[LayerName, kdb.Shapes] = defaultdict(kdb.Shapes) 

121 polygon_count = 0 

122 for region_by_layer in regions_within.values(): 

123 for layer_name, region in region_by_layer.items(): 

124 for polygon in region.merged().each(): 

125 self.shapes_by_layer[layer_name].insert(kdb.PolygonWithProperties(polygon.downcast(), 

126 {'id': polygon_count})) 

127 polygon_count += 1 

128 

129 self.polygon_sets = DisjointSets(range(polygon_count)) 

130 for bottom_layer_name, top_layer_name, vias in vias_within: 

131 for via in vias.each(): 

132 box = via.bbox() 

133 bottom, top = self.polygon_at(bottom_layer_name, box), self.polygon_at(top_layer_name, box) 

134 if bottom is not None and top is not None: 

135 self.polygon_sets.union(bottom, top)

137 def polygon_at(self, layer_name: LayerName, box: kdb.Box) -> Optional[int]: 

138 """ 

139 :return: the ID of the polygon on the layer with the largest overlap with the box 

140 """ 

141 area_by_id: Dict[int, int] = {} 

142 box_region = kdb.Region(box) 

143 for shape in self.shapes_by_layer[layer_name].each_touching(kdb.Shapes.SRegions, box): 

144 area = (kdb.Region(shape.polygon) & box_region).area() 

145 if area > 0: 

146 area_by_id[shape.property('id')] = area 

147 return max(sorted(area_by_id), key=area_by_id.get, default=None) 

148 

149 def set_at(self, layer_name: LayerName, box: kdb.Box) -> Optional[int]: 

150 """ 

151 :return: the ID of what is joined to the shapes at the box on the layer, None if there are no shapes 

152 """ 

153 polygon = self.polygon_at(layer_name, box) 

154 return None if polygon is None else self.polygon_sets.find(polygon) 

155 

156 def set_at_position(self, layer_name: LayerName, position: kdb.Point) -> Optional[int]: 

157 """ 

158 :return: the ID of what is joined to the shapes at the position (e.g. of a pin) on the layer, 

159 None if there are no shapes 

160 """ 

161 box = kdb.Box(position, position).enlarged(1, 1) 

162 # NOTE: where the metal stays for a via from above, what is around it has the position 

163 kept = self.kept_for_vias.get(layer_name, None) 

164 if kept is not None: 

165 for via in kept.interacting(kdb.Region(box)).each(): 

166 box = via.bbox().enlarged(1, 1) 

167 return self.set_at(layer_name, box)

201def label_names_net(label: Label, net_name: NetName) -> bool: 

202 """ 

203 :return: whether KLayout named the net after the label, regardless of case, as SPICE reads names: 

204 A of A,B (a net of several labels), vss of vss$1 (islands of one name), 

205 VDD of $1.VDD (a net within a subcircuit, which keeps that name where the label isn't unique, 

206 see KLayoutExtractionContext.name_nets_after_subcell_labels) 

207 """ 

208 label = label.casefold() 

209 for name in net_name.casefold().split(','): 

210 for candidate in (name, re.sub(r'\$\d+$', '', name)): 

211 if candidate == label or candidate.endswith(f".{label}"): 

212 return True 

213 return False

216def keep_pins_of_top_circuit(net_name: NetName, 

217 vertex_ports: Dict[int, List[kdb.Point]], 

218 vertex_port_pins: Dict[int, List[Tuple[Label, NetName]]]): 

219 """ 

220 Keeps the labels of the hierarchy that are pins of the top circuit, like the ports of its LVS netlist: 

221 a label in a subcell is none on a net that KLayout named after the labels of its own circuit (#403), 

222 and the first one of a net that no label names (e.g. with labels in subcells only) is the node of its port 

223 

224 NOTE: after the split at the exclusions, as a label within a marker names the port of the model there, 

225 also in a subcell (e.g. of an inductor) 

226 """ 

227 labels = {pin[0] for pins in vertex_port_pins.values() for pin in pins} 

228 if not labels: 

229 return 

230 pin_labels = {label for label in labels if label_names_net(label, net_name)} or {min(labels)} 

231 for layer_id in list(vertex_port_pins): 

232 kept = [(point, pin) 

233 for point, pin in zip(vertex_ports[layer_id], vertex_port_pins[layer_id]) 

234 if pin[0] in pin_labels] 

235 if kept: 

236 vertex_ports[layer_id] = [point for point, _ in kept] 

237 vertex_port_pins[layer_id] = [pin for _, pin in kept] 

238 else: 

239 del vertex_ports[layer_id] 

240 del vertex_port_pins[layer_id]

243def port_of_cut(strip: kdb.PolygonWithProperties, 

244 conductor_near: Callable[[kdb.Box], kdb.Region], 

245 strip_width: int, 

246 port_length: int) -> kdb.PolygonWithProperties: 

247 """ 

248 A polygon port is one node. So it's the whole strip where a conductor enters a block, across the current, 

249 also of a wide wire. But where the conductor goes on beyond the ends of the strip, the strip is along the current, 

250 beside metal that it would short (a port can't stand for a model in parallel to that metal), 

251 so then it's a short piece in the middle of the longest edge of the strip 

252 

253 :param strip: of a conductor along the cut 

254 :param conductor_near: the polygons of the conductor at a box 

255 :param strip_width: of the strip 

256 :param port_length: of the short piece, in DBU 

257 """ 

258 box = strip.bbox() 

259 if box.width() >= box.height(): 

260 ends = (kdb.Box(box.left - 1, box.bottom, box.left, box.top), 

261 kdb.Box(box.right, box.bottom, box.right + 1, box.top)) 

262 else: 

263 ends = (kdb.Box(box.left, box.bottom - 1, box.right, box.bottom), 

264 kdb.Box(box.left, box.top, box.right, box.top + 1)) 

265 if not strip.is_rectilinear() or all((conductor_near(end) & kdb.Region(end)).is_empty() for end in ends): 

266 return strip 

267 

268 edge = max(strip.each_edge(), key=lambda e: e.length()) 

269 if edge.length() <= port_length: 

270 return strip 

271 middle = kdb.Point((edge.x1 + edge.x2) // 2, (edge.y1 + edge.y2) // 2) 

272 half_length = port_length // 2 

273 window = kdb.Box(middle.x - half_length, middle.y - strip_width, middle.x + half_length, middle.y + strip_width) \ 

274 if edge.dy() == 0 else \ 

275 kdb.Box(middle.x - strip_width, middle.y - half_length, middle.x + strip_width, middle.y + half_length) 

276 return next((kdb.Region(strip.downcast()) & kdb.Region(window)).each())

314 def prepare_r_extractor_tech_pb(self, 

315 rex_tech: pb_RExtractorTech): 

316 """ 

317 Prepare KLayout PEXCore Technology Description based on the KPEX Tech Info data 

318 :param rex_tech: RExtractorTech protobuffer message 

319 """ 

320 

321 rex_tech.skip_simplify = self.skip_simplify 

322 

323 tech = self.pex_context.tech 

324 

325 # NOTE: a layer that can't be modeled is not part of the resistance network, 

326 # e.g. a via that joins nothing, which splits the networks of its nets. 

327 # This is never intended, so it's an error rather than a warning (#217). 

328 # It includes the LVS layers the tech info has no layer for (e.g. a via with another name) 

329 unmodeled_layers: List[str] = list(self.pex_context.unmodeled_layers) 

330 

331 # NOTE: the resistance extraction works on the layer of a GDS pair, with the shapes of all its LVS layers, 

332 # so one model covers them all (e.g. sky130A mcon_con and mcon_vpp, the vias within MOM caps) 

333 unmodeled_layers_by_gds_pair: Dict[GDSPair, List[str]] = defaultdict(list) 

334 modeled_gds_pairs: Set[GDSPair] = set() 

335 

336 substrate_gds_pairs = {tech.gds_pair_for_computed_layer_name[name] 

337 for name in tech.tech.substrate.lvs_layer_names 

338 if name in tech.gds_pair_for_computed_layer_name} 

339 wells_without_resistance: Set[str] = set() 

340 

341 for gds_pair, li in self.pex_context.extracted_layers.items(): 

342 for source_layer in li.source_layers: 

343 computed_layer_info = tech.computed_layer_info_by_name.get(source_layer.lvs_layer_name, None) 

344 if computed_layer_info is None: 

345 unmodeled_layers_by_gds_pair[gds_pair].append(f"LVS {source_layer.lvs_layer_name} {gds_pair}: " 

346 f"no computed layer info") 

347 continue 

348 

349 canonical_layer_name = tech.canonical_layer_name_by_gds_pair[gds_pair] 

350 layer_description = f"{canonical_layer_name} (LVS {source_layer.lvs_layer_name})" 

351 

352 LP = tech_pb2.LayerInfo.Purpose 

353 

354 match computed_layer_info.kind: 

355 case tech_pb2.ComputedLayerInfo.Kind.KIND_PIN: 

356 continue 

357 

358 case tech_pb2.ComputedLayerInfo.Kind.KIND_LABEL: 

359 continue 

360 

361 case _: 

362 pass 

363 

364 match computed_layer_info.layer_info.purpose: 

365 case LP.PURPOSE_NWELL | LP.PURPOSE_PWELL: 

366 # NOTE: a well is a conductor between the contacts over its taps, which land on the well 

367 # (e.g. sky130A licon_ntap_con), and the device terminals on it (e.g. the PMOS bulks), 

368 # but the substrate (e.g. IHP pwell_sub, on the GDS pair of pwell), which is ideal, 

369 # as it is 3D bulk rather than a sheet, so the device terminals on it stay on 

370 # the node of their net 

371 if gds_pair in substrate_gds_pairs: 

372 continue 

373 

374 layer_resistance = tech.layer_resistance_by_layer_name.get(canonical_layer_name, None) 

375 if layer_resistance is None: 

376 wells_without_resistance.add(layer_description) 

377 continue 

378 

379 cond = rex_tech.conductors.add() 

380 

381 cond.layer.id = self.pex_context.annotated_layout.layer(*source_layer.gds_pair) 

382 cond.layer.canonical_layer_name = canonical_layer_name 

383 cond.layer.lvs_layer_name = source_layer.lvs_layer_name 

384 

385 cond.triangulation_min_b = self.delaunay_b 

386 cond.triangulation_max_area = self.delaunay_amax 

387 

388 cond.algorithm = self.substrate_algorithm 

389 cond.resistance = self.pex_context.tech.milliohm_to_ohm(layer_resistance.resistance) 

390 modeled_gds_pairs.add(gds_pair) 

391 

392 case LP.PURPOSE_DNWELL: 

393 # NOTE: no conductor: no via joins the deep nwell to the nwell over it, 

394 # so it would be a piece that joins nothing (see isolated_well_node_ids), 

395 # and gf180mcuD has no sheet resistance for it. Like the substrate, 

396 # the device terminals on it stay on the node of their net (#292) 

397 continue 

398 

399 case LP.PURPOSE_NTAP | LP.PURPOSE_PTAP: 

400 # NOTE: no conductor, the contacts over a tap land on the well below it 

401 # (e.g. sky130A licon_ntap_con on the nwell), neglecting the resistance 

402 # of the tap itself, like the one of the diffusion (e.g. SkyWater rule SR.diff.1) 

403 continue 

404 

405 case LP.PURPOSE_N_IMPLANT | LP.PURPOSE_P_IMPLANT: 

406 # device terminals 

407 # - source/drain (e.g. sky130A: nsdm, psdm) 

408 # - bulk (e.g. nwell) 

409 # 

410 # we will consider this only as a pin end-point, there are no wires at all on this layer, 

411 # so the resistance does not matter for PEX 

412 cond = rex_tech.conductors.add() 

413 

414 cond.layer.id = self.pex_context.annotated_layout.layer(*source_layer.gds_pair) 

415 cond.layer.canonical_layer_name = canonical_layer_name 

416 cond.layer.lvs_layer_name = source_layer.lvs_layer_name 

417 

418 cond.triangulation_min_b = self.delaunay_b 

419 cond.triangulation_max_area = self.delaunay_amax 

420 

421 cond.algorithm = self.substrate_algorithm 

422 cond.resistance = 0 # see comment above 

423 modeled_gds_pairs.add(gds_pair) 

424 

425 case LP.PURPOSE_METAL | LP.PURPOSE_MIM_CAP: 

426 # NOTE: the top plate of a MIM cap is a thin metal, which the white-box extraction keeps 

427 # (the MIM cap layers are no part of the extraction in the black-box mode) 

428 layer_resistance = tech.layer_resistance_by_layer_name.get(canonical_layer_name, None) 

429 if layer_resistance is None: 

430 unmodeled_layers_by_gds_pair[gds_pair].append(f"{layer_description}: no sheet resistance") 

431 continue 

432 

433 cond = rex_tech.conductors.add() 

434 

435 cond.layer.id = self.pex_context.annotated_layout.layer(*source_layer.gds_pair) 

436 cond.layer.canonical_layer_name = canonical_layer_name 

437 cond.layer.lvs_layer_name = source_layer.lvs_layer_name 

438 

439 cond.triangulation_min_b = self.delaunay_b 

440 cond.triangulation_max_area = self.delaunay_amax 

441 

442 if canonical_layer_name == tech.internal_substrate_layer_name: 

443 cond.algorithm = self.substrate_algorithm 

444 else: 

445 cond.algorithm = self.wire_algorithm 

446 cond.resistance = self.pex_context.tech.milliohm_to_ohm(layer_resistance.resistance) 

447 modeled_gds_pairs.add(gds_pair) 

448 

449 case LP.PURPOSE_CONTACT: 

450 contact = tech.contact_by_contact_lvs_layer_name.get(source_layer.lvs_layer_name, None) 

451 if contact is None: 

452 unmodeled_layers_by_gds_pair[gds_pair].append(f"{layer_description}: no contact") 

453 continue 

454 

455 contact_resistance = tech.contact_resistance_by_device_layer_name.get(contact.layer_below, 

456 None) 

457 if contact_resistance is None: 

458 unmodeled_layers_by_gds_pair[gds_pair].append( 

459 f"{layer_description}: no contact resistance for {contact.layer_below}") 

460 continue 

461 

462 via = rex_tech.vias.add() 

463 

464 bot_gds_pair = tech.gds_pair(contact.layer_below) 

465 top_gds_pair = tech.gds_pair(contact.metal_above) 

466 

467 via.layer.id = self.pex_context.annotated_layout.layer(*source_layer.gds_pair) 

468 via.layer.canonical_layer_name = canonical_layer_name 

469 via.layer.lvs_layer_name = source_layer.lvs_layer_name 

470 

471 via.bottom_conductor.id = self.pex_context.annotated_layout.layer(*bot_gds_pair) 

472 via.top_conductor.id = self.pex_context.annotated_layout.layer(*top_gds_pair) 

473 

474 via.resistance = self.pex_context.tech.milliohm_by_cnt_to_ohm_by_square_for_contact( 

475 contact=contact, 

476 contact_resistance=contact_resistance 

477 ) 

478 via.merge_distance = self.via_merge_distance 

479 modeled_gds_pairs.add(gds_pair) 

480 

481 case LP.PURPOSE_VIA: 

482 via_resistance = tech.via_resistance_by_layer_name.get(canonical_layer_name, None) 

483 if via_resistance is None: 

484 unmodeled_layers_by_gds_pair[gds_pair].append(f"{layer_description}: no via resistance") 

485 continue 

486 # NOTE: the via of the process stack on the GDS pair, if it has none named after the LVS layer 

487 # (e.g. the vias of a device, which the LVS deck has apart from the others) 

488 via_name = tech.process_layer_name_by_lvs_layer_name.get(source_layer.lvs_layer_name, 

489 source_layer.lvs_layer_name)

681 def split_at_exclusions(self, 

682 tech: pb_RExtractorTech, 

683 regions: Dict[int, kdb.Region], 

684 vertex_ports: Dict[int, List[kdb.Point]], 

685 vertex_port_pins: Dict[int, List[Tuple[Label, NetName]]], 

686 polygon_ports: Dict[int, List[kdb.Polygon]], 

687 polygon_port_device_terminals: PolygonPortTerminals, 

688 layer_names: Dict[int, LayerName]) -> NetSplit: 

689 """ 

690 Takes the shapes of a net within the exclusion markers out of its network. 

691 A polygon port on the conductor beside each cut is its node, as a whole cut is one node, also of a wide wire. 

692 

693 Either these become ports of the netlist, for the model of what the markers stand for, 

694 like the device terminals within the markers, which the model connects then. 

695 

696 Or the shapes go to layers without resistance (ideal_layer_id), so that the extraction still tells what they 

697 connect, with their pins and device terminals, and a polygon port on them joins each cut to them. 

698 

699 NOTE: a via is within a block by its center, which is the location of its nodes, on both of its conductors, 

700 otherwise it keeps its resistance and the metal it lands on, as a via port needs its center on the metal, 

701 and the resistance of a part of it would be made up 

702 

703 :param tech: the conductors and vias, with their ideal copies 

704 :param regions: the shapes of the net by layer ID 

705 :param vertex_ports: the pins by layer ID 

706 :param vertex_port_pins: what the pins are of, by layer ID 

707 :param polygon_ports: the device terminals by layer ID 

708 :param polygon_port_device_terminals: what the polygon ports are of 

709 :param layer_names: the canonical names of the conductors and vias, by layer ID 

710 """ 

711 split = NetSplit() 

712 exclusions = self.exclusions 

713 

714 def is_ideal(layer_id: int) -> bool: 

715 return layer_id >= IDEAL_LAYER_ID_OFFSET 

716 

717 # NOTE: the device terminals of the removed devices are no ports of devices anymore 

718 removed_terminals: List[Tuple[int, kdb.Polygon, device_pb2.Device.Terminal]] = [] 

719 removed_terminal_shapes: Dict[int, kdb.Shapes] = defaultdict(kdb.Shapes) 

720 for layer_id in list(polygon_ports): 

721 ports_and_terminals = list(zip(polygon_ports.pop(layer_id), polygon_port_device_terminals.pop(layer_id))) 

722 for port, terminal in ports_and_terminals: 

723 if terminal.device_id in exclusions.removed_device_blocks: 

724 removed_terminal_shapes[layer_id].insert(kdb.PolygonWithProperties( 

725 port, {'terminal': len(removed_terminals)})) 

726 removed_terminals.append((layer_id, port, terminal)) 

727 else: 

728 polygon_ports[layer_id].append(port) 

729 polygon_port_device_terminals[layer_id].append(terminal) 

730 

731 kept_for_vias: Dict[int, kdb.Region] = defaultdict(kdb.Region) 

732 for via in {v.layer.id: v for v in tech.vias if not is_ideal(v.layer.id)}.values(): 

733 shapes = regions.get(via.layer.id, None) 

734 # NOTE: most nets are far from the blocks 

735 if shapes is None or not shapes.bbox().touches(exclusions.bbox): 

736 continue 

737 bottom_layer_name = layer_names[via.bottom_conductor.id] 

738 top_layer_name = layer_names[via.top_conductor.id] 

739 # NOTE: within a block on both of its conductors 

740 box = shapes.bbox() 

741 within_tie = polygons_by_center_inside(shapes, exclusions.mask_near(box, bottom_layer_name, ports=False) & 

742 exclusions.mask_near(box, top_layer_name, ports=False)) 

743 within_ports = polygons_by_center_inside(shapes, exclusions.mask_near(box, bottom_layer_name, ports=True) & 

744 exclusions.mask_near(box, top_layer_name, ports=True)) 

745 if not within_tie.is_empty(): 

746 regions[ideal_layer_id(via.layer.id)] = within_tie 

747 if not within_ports.is_empty(): 

748 split.vias_within.append((bottom_layer_name, top_layer_name, within_ports)) 

749 rest = shapes - within_tie - within_ports 

750 regions[via.layer.id] = rest 

751 across = rest.interacting(exclusions.mask_near(box, bottom_layer_name) + 

752 exclusions.mask_near(box, top_layer_name)) 

753 kept_for_vias[via.bottom_conductor.id] += across 

754 kept_for_vias[via.top_conductor.id] += across 

755 

756 for layer_id in sorted({c.layer.id for c in tech.conductors if not is_ideal(c.layer.id)}): 

757 shapes = regions.get(layer_id, None) 

758 if shapes is None or not shapes.bbox().touches(exclusions.bbox): 

759 continue 

760 layer_name = layer_names[layer_id] 

761 box = shapes.bbox() 

762 mask_ports = exclusions.mask_near(box, layer_name, ports=True) - kept_for_vias[layer_id] 

763 mask_tie = exclusions.mask_near(box, layer_name, ports=False) - kept_for_vias[layer_id] - mask_ports 

764 within_ports = shapes & mask_ports 

765 within_tie = shapes & mask_tie 

766 if within_tie.is_empty() and within_ports.is_empty(): 

767 continue 

768 # NOTE: without the masks rather than what is within them, as any-angle edges round differently, 

769 # which leaves slivers along the cuts otherwise, that join what is on both sides of a block 

770 outside = shapes - mask_ports - mask_tie 

771 ideal_id = ideal_layer_id(layer_id) 

772 regions[layer_id] = outside 

773 split.kept_for_vias[layer_name] = kept_for_vias[layer_id] 

774 if not within_tie.is_empty(): 

775 regions[ideal_id] = within_tie 

776 for polygon in within_ports.merged().each(): 

777 block_indexes = exclusions.block_indexes_with_ports_within(kdb.Region(polygon.downcast()), layer_name) 

778 if len(block_indexes) > 1: 

779 split.abutting_block_indexes.update(block_indexes) 

780 region_by_layer = split.regions_within[block_indexes[0]] 

781 region_by_layer.setdefault(layer_name, kdb.Region()).insert(polygon.downcast()) 

782 

783 # NOTE: the pins and device terminals go with the conductor they are on, those on both to both of them, 

784 # which are one node, as the ports of a device terminal are 

785 for point, pin in zip(vertex_ports.pop(layer_id, []), vertex_port_pins.pop(layer_id, [])): 

786 label_box = kdb.Region(kdb.Box(point, point).enlarged(1, 1)) 

787 # NOTE: a pin of a cut cell is none of the top cell, but it names the port of the model, 

788 # also where its metal stays for a via from above, which can land on it 

789 if exclusions.is_only_label_of_cut_cell(pin[0], point) and \ 

790 not (label_box & exclusions.mask_near(label_box.bbox(), layer_name, ports=True)).is_empty(): 

791 continue 

792 if not (label_box & outside).is_empty(): 

793 vertex_ports[layer_id].append(point) 

794 vertex_port_pins[layer_id].append(pin) 

795 elif not (label_box & within_ports).is_empty(): 

796 split.pins_within.append(PinWithinMarker(point, layer_id, *pin)) 

797 else: 

798 vertex_ports[ideal_id].append(point) 

799 vertex_port_pins[ideal_id].append(pin)

809 port_region = kdb.Region(port) 

810 part_tie = port_region & within_tie 

811 part_ports = port_region & within_ports

823 else ANY_ANGLE_CUT_STRIP_WIDTH

825

828 for strip_layer_id, conductor in ((layer_id, outside), (ideal_id, within_tie)): 

829 for strip in (conductor & around_ports).merged().each():

831 round(PORT_LENGTH_ALONG_CONDUCTOR / self.pex_context.dbu))

837 terminal = removed_terminals[shape.property('terminal')][2] 

838 if exclusions.removed_device_blocks[terminal.device_id] == split.cut_port_blocks[key] and \ 

839 not (kdb.Region(shape.polygon) & kdb.Region(port.downcast())).is_empty(): 

840 split.terminals_of_cuts[key] = terminal 

841 break 

842 for outside_strip in (outside & within_tie.sized(strip_width) - around_ports).merged().each(): 

843 outside_key = add_port(layer_id, outside_strip) 

844 for within_strip in (within_tie & kdb.Region(outside_strip).sized(strip_width)).merged().each(): 

845 split.polygon_port_ties.append((outside_key, add_port(ideal_id, within_strip))) 

846 

847 # NOTE: a removed device stands for what its model has, and where it meets the rest of its net is a port, 

848 # e.g. the terminal of a coil on its lead outside its marker 

849 conductor_layer_ids = {c.layer.id for c in tech.conductors} 

850 for layer_id, port, terminal in removed_terminals: 

851 shapes = regions.get(layer_id, None) 

852 if layer_id not in conductor_layer_ids or shapes is None or shapes.is_empty(): 

853 continue # NOTE: e.g. the bulk of a transistor on the substrate 

854 for polygon in (kdb.Region(port) & shapes).merged().each(): 

855 polygon_ports[layer_id].append(polygon.downcast()) 

856 polygon_port_device_terminals[layer_id].append(None) 

857 key = (layer_id, len(polygon_ports[layer_id]) - 1) 

858 split.cut_port_boxes[key] = polygon.bbox() 

859 split.cut_port_blocks[key] = exclusions.removed_device_blocks[terminal.device_id] 

860 split.cut_port_terminals[key] = terminal 

861 return split

925 for label in self.labels_at_shapes(block_index, split)] 

926 for block_index in set(split.cut_port_blocks.values())} 

927 for key in cut_keys: 

928 # NOTE: a port at the terminal of a removed device is named after that terminal 

929 if key in split.cut_port_terminals: 

930 continue 

931 block_index = split.cut_port_blocks[key] 

932 cut_box = split.cut_port_boxes[key] 

933 cut_set = sets_of_cuts.get(key, None) 

934 if cut_set is None: 

935 continue

937 if pin_set == cut_set and pin.label not in labels_outside and block_index_of_pin(pin) == block_index:

939 distance(pin.position, cut_box), 

940 pin.label)) 

941 for label, label_set in labels_with_sets[block_index]: 

942 if label_set == cut_set: 

943 candidates_by_cut[key].append((label.layer_name != layer_names[key[0]], 

944 distance(label.position, cut_box), 

945 f"{blocks[block_index].name}__{label.text}"))

947 # NOTE: the nearest pairs first, on the layer of the pin before others, each pin names one cut 

948 names_by_cut: Dict[PolygonPortKey, str] = {} 

949 names_left = {candidate[2] for candidates in candidates_by_cut.values() for candidate in candidates} 

950 for _, _, name, _, key in sorted((*candidate, index, key) 

951 for index, key in enumerate(cut_keys) 

952 for candidate in candidates_by_cut[key]): 

953 if name in names_left and key not in names_by_cut: 

954 names_by_cut[key] = name 

955 names_left.discard(name) 

956 

957 numbers: Counter[int] = Counter() 

958 taken_names: Set[str] = {candidate[2] for candidates in candidates_by_cut.values() for candidate in candidates} 

959 taken_names.update(labels_outside) 

960 taken_names.update(pin.label for pin in split.pins_within)

962 def numbered_name(block_index: int) -> str: 

963 while True: 

964 number = numbers[block_index] 

965 numbers[block_index] += 1 

966 name = f"{blocks[block_index].name}__{safe_net_name}_{number}" \ 

967 if blocks[block_index].kind == BlockKind.CUT else \ 

968 f"{safe_net_name}__{blocks[block_index].name}_{number}" 

969 if name not in taken_names: 

970 taken_names.add(name) 

971 return name 

972 

973 for key in cut_keys: 

974 if key in names_by_cut: 

975 continue 

976 block_index = split.cut_port_blocks[key] 

977 cut_candidates = [c for c in candidates_by_cut[key] if c[2].startswith(f"{blocks[block_index].name}__")] 

978 if blocks[block_index].kind == BlockKind.CUT and cut_candidates: 

979 names_by_cut[key] = min(cut_candidates)[2] 

980 elif key not in split.cut_port_terminals and key not in split.terminals_of_cuts: 

981 names_by_cut[key] = numbered_name(block_index) 

982 

983 names_by_cut.update(self.terminal_port_names(split, [key for key in cut_keys if key not in names_by_cut], 

984 taken_names))

1017 def terminal_port_names(self, 

1018 split: NetSplit, 

1019 keys: Sequence[PolygonPortKey], 

1020 taken_names: Set[str]) -> Dict[PolygonPortKey, str]: 

1021 """ 

1022 A port is one node: of a terminal of a device, also of several polygons, 

1023 and of the terminals of that name of the other devices of the block on the same polygon 

1024 

1025 :param keys: the polygon ports at the terminals of removed devices 

1026 :param taken_names: the names of the other ports of the net, to which these go too 

1027 :return: their names, <block>__<terminal>, numbered if the block has several such nodes on the net 

1028 """ 

1029 blocks = self.exclusions.blocks 

1030 terminals = {**split.terminals_of_cuts, **split.cut_port_terminals} 

1031 ports = DisjointSets(keys) 

1032 first_key_by_node: Dict[Hashable, PolygonPortKey] = {} 

1033 for key in keys: 

1034 terminal = terminals[key] 

1035 for node in ((terminal.device_id, terminal.terminal_id), 

1036 (split.cut_port_blocks[key], terminal.name, key[0], split.cut_port_boxes[key])): 

1037 ports.union(key, first_key_by_node.setdefault(node, key)) 

1038 

1039 roots_by_name: Dict[str, List[PolygonPortKey]] = defaultdict(list) 

1040 for key in keys: 

1041 name = f"{blocks[split.cut_port_blocks[key]].name}__{terminals[key].name}" 

1042 root = ports.find(key) 

1043 if root not in roots_by_name[name]: 

1044 roots_by_name[name].append(root) 

1045 

1046 name_by_root: Dict[PolygonPortKey, str] = {} 

1047 for name, roots in roots_by_name.items(): 

1048 if len(roots) == 1 and name not in taken_names: 

1049 name_by_root[roots[0]] = name 

1050 taken_names.add(name) 

1051 continue 

1052 numbers = (f"{name}_{number}" for number in itertools.count()) 

1053 for root in roots: 

1054 name_by_root[root] = next(number for number in numbers if number not in taken_names) 

1055 taken_names.add(name_by_root[root]) 

1056 return {key: name_by_root[ports.find(key)] for key in keys}

1058 def internal_node_indexes(self, 

1059 resistor_network: klp.RNetwork, 

1060 layer_names: Dict[int, LayerName], 

1061 excluded_node_ids: Set[int]) -> Dict[int, int]: 

1062 """ 

1063 The index of each internal node of a resistor network, by its object ID: 

1064 in the order of their layers and locations, nodes of the same location by their elements 

1065 

1066 NOTE: KLayout numbers the internal nodes in the order it creates them, which can differ from run to run 

1067 (e.g. KLayout 0.30.12), and their names decide ties in the distribution of the capacitances (#286) 

1068 """ 

1069 dbu = self.pex_context.dbu 

1070 

1071 def location(rn: klp.RNode) -> Tuple[LayerName, str, int, int, int, int, int]: 

1072 if rn.type() != klp.RNodeType.Internal: 

1073 # NOTE: a port by its index, as KLayout's location of a vertex port can differ from run to run 

1074 return layer_names[rn.layer()], str(rn.type()), rn.port_index(), 0, 0, 0, 0 

1075 box = rn.location() 

1076 p1 = box.p1.to_itype(dbu) 

1077 p2 = box.p2.to_itype(dbu) 

1078 return layer_names[rn.layer()], str(rn.type()), -1, p1.x, p1.y, p2.x, p2.y 

1079 

1080 neighbors: Dict[int, List[Tuple[float, Tuple]]] = defaultdict(list) 

1081 for el in resistor_network.each_element(): 

1082 neighbors[el.a().object_id()].append((el.resistance(), location(el.b()))) 

1083 neighbors[el.b().object_id()].append((el.resistance(), location(el.a()))) 

1084 

1085 keys = {rn.object_id(): (location(rn), sorted(neighbors[rn.object_id()])) 

1086 for rn in resistor_network.each_node() 

1087 if rn.type() == klp.RNodeType.Internal and rn.object_id() not in excluded_node_ids} 

1088 return {node_id: index for index, node_id in enumerate(sorted(keys, key=keys.get))}

1444 def with_substrate(region_by_layer: Dict[LayerName, kdb.Region]) -> Dict[LayerName, kdb.Region]:

1446 wells = [region for layer_name, region in region_by_layer.items() if layer_name in well_layer_names]
